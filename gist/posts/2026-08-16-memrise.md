## Javascript

### Memrise Japanese Context Sentences
```javascript
// ==UserScript==
// @name           Memrise Japanese Context Sentences
// @description    Shows real-media example sentences (image + audio, via Immersion Kit) for the Japanese word currently on screen during a Memrise learn/review session. After you answer correctly, one pops up full-size in its own card and the session pauses so you can hear the audio in full before continuing.
// @match          https://community-courses.memrise.com/*
// @match          https://community-courses.memrise.com/aprender/*
// @match          https://*.memrise.com/aprender/*
// @match          https://*.memrise.com/course/*/garden/*
// @match          https://*.memrise.com/garden/review/*
// @run-at         document-end
// @version        1.8.0
// @grant          none
// @namespace      https://greasyfork.org/users/213706
// ==/UserScript==

// Sentence/audio/image data comes from the free Immersion Kit API
// (https://www.immersionkit.com) - the same public search endpoint used by
// scripts like "WaniKani Media Context Sentences". This is an independent,
// from-scratch implementation for Memrise; it shares no code with that
// script, just the same public data source.
//
// v1.1.0 changes:
//  - Everything now attaches to <html> (document.documentElement) instead
//    of <body>. Memrise's React app likely wraps its content in a container
//    with a CSS transform (common for page-transition animations), and a
//    position:fixed element becomes fixed *relative to that ancestor*
//    instead of the viewport - which can push it off-screen or clip it
//    entirely. That's the most likely reason nothing was visible before.
//  - Both the settings toggle and the example box now sit bottom-LEFT
//    (matching the Memrise Audio Provider script's corner), stacked with
//    enough clearance above it that the two shouldn't overlap when both
//    are installed.
//  - The in-session box now shows MULTIPLE example sentences (a scrollable
//    list), not just one. Only the first one autoplays; the rest play on
//    click, same as before.
//  - Added console diagnostics at every startup stage (look for lines
//    starting "Context Sentences:") plus a visibility self-check, so if
//    something still doesn't show up, what's logged should say why.
//
// HOW "PAUSING" WORKS: Memrise's own auto-advance (if the round has one) is
// timer-based internally and isn't something a userscript can reach, so
// this can only ever *block user interaction* with the page - it swallows
// clicks/keydowns in the capture phase while the card is open. That's
// enough in practice (nothing you do reaches Memrise's own handlers), but
// if a build ever auto-advances on its own after some fixed delay
// regardless of input, that's outside what this script can prevent.
//
// CORRECT-ANSWER DETECTION IS HEURISTIC. Memrise's front-end doesn't expose
// a stable "you got it right" event/attribute, so this watches for the same
// kind of visual signals the Memrise Audio Provider script uses (a green
// highlight on the chosen multiple-choice button, a green-bordered typed
// answer, or a generic "Correct!"-style toast). If it doesn't fire on some
// round type you use, tell me what the DOM looks like right after
// answering and I'll add a case for it.

(function () {

  if (window.__memriseContextSentencesLoaded) { console.log("Context Sentences: already loaded, skipping"); return; }
  window.__memriseContextSentencesLoaded = true;
  console.log("Context Sentences: script starting");

  const JAPANESE_CHAR_REGEX = /[\u3040-\u30FF\u3400-\u4DBF\u4E00-\u9FFF\uFF66-\uFF9F]/;

  const IMMERSIONKIT_ORIGIN = "https://apiv2.immersionkit.com";
  const IMMERSIONKIT_SEARCH_PATH = "/search";
  const IMMERSIONKIT_INDEX_PATH = "/index_meta";
  // Media CDN layout, VERIFIED against the live service:
  //   {base}/{category}/{Pretty Title}/media/{filename}
  // Note it is the PRETTY title ("Clannad"), not the slug the search
  // endpoint returns ("clannad") - requesting the slug form returns 403.
  const IMMERSIONKIT_MEDIA_BASE = "https://us-southeast-1.linodeobjects.com/immersionkit/media/";

  const LOCALSTORAGE_SETTINGS = "memrise-context-sentences-settings",
        LOCALSTORAGE_CACHE    = "memrise-context-sentences-cache",
        LOCALSTORAGE_DECKMETA = "memrise-context-sentences-deckmeta";

  const DEFAULT_PROMPT_SELECTORS = [
    '[data-testid$="prompt-text"]',
    '[data-testid="prompt-text"]', '[data-testid="prompt"]',
    '.prompt .text', '.test-prompt .text', '.prompt-text',
    '.column .text', '#prompt .text'
  ];

  const DEFAULT_SETTINGS = {
    enabled: true,
    showInlineBox: true,
    autoplayFirstExample: true,
    showOnPresentation: true,
    showOnCorrect: true,
    showOnIncorrect: true,
    autoplayOnCorrect: true,
    exampleLimit: 5,
  };

  function storageGet(key, fallback) {
    try { const raw = localStorage.getItem(key); return raw === null ? fallback : JSON.parse(raw); }
    catch (e) { return fallback; }
  }
  function storageSet(key, value) {
    try { localStorage.setItem(key, JSON.stringify(value)); } catch (e) { /* private mode */ }
  }
  function log(msg) { console.log("Context Sentences: " + msg); }
  function warn(msg) { console.warn("Context Sentences: " + msg); }
  function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c];
    });
  }
  function containsJapanese(text) { return !!text && JAPANESE_CHAR_REGEX.test(text); }
  // Everything gets appended here rather than document.body - see the
  // v1.1.0 note above about transformed ancestors breaking position:fixed.
  function rootContainer() { return document.documentElement; }

  var CS = {

    init: function () {
      if (this.initialised) { return; }
      this.initialised = true;
      log("init() running");

      this.settings = Object.assign({}, DEFAULT_SETTINGS, storageGet(LOCALSTORAGE_SETTINGS, {}));
      this.cache    = storageGet(LOCALSTORAGE_CACHE, {}); // word -> {examples: [...], fetchedAt}

      this.mode            = "waiting"; // waiting | memrise | dom
      this.currentWord     = "";
      this.currentExamples = undefined; // undefined = not fetched yet, [] = fetched, no results
      this.autoplayIndex   = null; // which side-panel card autoplays (random per word)
      this.fetchToken      = 0;
      this.injectedFor     = null; // word the currently-injected example belongs to, or null
      this.injectedAnchor  = null; // element it was attached to, watched for removal
      this.injectedMisses  = 0;
      this.inlineAudioEls  = [];
      this.pendingInflight = {};
      this.deckIndex       = null; // slug -> {title, category, tags}, from /index_meta
      this.deckIndexPromise = null;

      try { this.loadDeckIndex(); } catch (e) { warn("loadDeckIndex() threw: " + e.message); }
      try { this.buildPanel(); } catch (e) { warn("buildPanel() threw: " + e.message); }
      try { this.watchSession(); } catch (e) { warn("watchSession() threw: " + e.message); }
      try { this.watchForCorrectAnswer(); } catch (e) { warn("watchForCorrectAnswer() threw: " + e.message); }

      log("init() finished - window.MemriseContextSentences.diagnose() for details");
      setTimeout(function () { CS.selfCheckVisibility(); }, 800);
    },

    /** Logs whether the panel actually has non-zero, on-screen dimensions, so a "nothing shows up" report can be diagnosed from the console alone. */
    selfCheckVisibility: function () {
      var panel = document.getElementById('context-sentences-panel');
      if (!panel) { warn("settings panel is not even in the DOM - buildPanel() likely failed, see earlier warnings"); return; }
      var rect = panel.getBoundingClientRect();
      var style = window.getComputedStyle(panel);
      log("settings panel rect=" + JSON.stringify({ w: rect.width, h: rect.height, top: rect.top, left: rect.left })
        + " display=" + style.display + " visibility=" + style.visibility + " zIndex=" + style.zIndex);
      if (rect.width === 0 || rect.height === 0) {
        warn("settings panel has zero size - something on the page is likely overriding its CSS. Check the Elements/Inspector tab for #context-sentences-panel.");
      }
      if (rect.top < 0 || rect.left < 0 || rect.top > window.innerHeight || rect.left > window.innerWidth) {
        warn("settings panel rect is outside the viewport - likely a transformed ancestor is capturing position:fixed. Check computed style of ancestors of <html> > #context-sentences-panel.");
      }
    },

    //+------------------------------------------------------
    //| SETTINGS / PANEL
    //+------------------------------------------------------

    saveSettings: function () { storageSet(LOCALSTORAGE_SETTINGS, this.settings); },

    buildPanel: function () {
      var s = this.settings;
      var div = document.createElement('div');
      div.id = "context-sentences-panel";
      div.innerHTML = `
        <p id="cs-link" style="cursor:pointer; margin:0; font-weight:bold;">Context Sentences &#9662;</p>
        <div id="cs-box" style="display:none; padding:6px 0 0;">
          <label style="display:block;"><input id="cs-enabled" type="checkbox" ${s.enabled ? "checked" : ""}> Enabled</label>
          <label style="display:block;"><input id="cs-inline" type="checkbox" ${s.showInlineBox ? "checked" : ""}> Show box while learning</label>
          <label style="display:block;"><input id="cs-autoplayfirst" type="checkbox" ${s.autoplayFirstExample ? "checked" : ""}> Autoplay one example's audio (random)</label>
          <label style="display:block;"><input id="cs-onpresentation" type="checkbox" ${s.showOnPresentation ? "checked" : ""}> Show on new-word screen</label>
          <label style="display:block;"><input id="cs-oncorrect" type="checkbox" ${s.showOnCorrect ? "checked" : ""}> Show when answered correctly</label>
          <label style="display:block;"><input id="cs-onincorrect" type="checkbox" ${s.showOnIncorrect ? "checked" : ""}> Show when answered incorrectly</label>
          <label style="display:block;"><input id="cs-oncorrect-audio" type="checkbox" ${s.autoplayOnCorrect ? "checked" : ""}> Autoplay that example's audio</label>
          <div id="cs-status" style="font-size:80%; margin-top:6px; color:#ccc;"></div>
        </div>`;
      rootContainer().appendChild(div);
      log("settings panel appended to " + (rootContainer() === document.documentElement ? "<html>" : "<body>"));

      var $ = function (id) { return document.getElementById(id); };
      $('cs-link').addEventListener('click', function () {
        var box = $('cs-box');
        box.style.display = (box.style.display === "none" ? "block" : "none");
      });
      $('cs-enabled').addEventListener('change', function () { CS.settings.enabled = this.checked; CS.saveSettings(); CS.refreshInlineBox(); });
      $('cs-inline').addEventListener('change', function () { CS.settings.showInlineBox = this.checked; CS.saveSettings(); CS.refreshInlineBox(); });
      $('cs-autoplayfirst').addEventListener('change', function () { CS.settings.autoplayFirstExample = this.checked; CS.saveSettings(); });
      $('cs-onpresentation').addEventListener('change', function () { CS.settings.showOnPresentation = this.checked; CS.saveSettings(); });
      $('cs-oncorrect').addEventListener('change', function () { CS.settings.showOnCorrect = this.checked; CS.saveSettings(); });
      $('cs-onincorrect').addEventListener('change', function () { CS.settings.showOnIncorrect = this.checked; CS.saveSettings(); });
      $('cs-oncorrect-audio').addEventListener('change', function () { CS.settings.autoplayOnCorrect = this.checked; CS.saveSettings(); });
    },

    setStatus: function (text) {
      var el = document.getElementById('cs-status');
      if (el) { el.textContent = text; }
    },

    diagnose: function () {
      var info = {
        version: "1.8.0", mode: this.mode, currentWord: this.currentWord,
        exampleCount: Array.isArray(this.currentExamples) ? this.currentExamples.length : null,
        settings: this.settings, injectedFor: this.injectedFor,
        cacheSize: Object.keys(this.cache).length,
        deckIndexLoaded: !!this.deckIndex,
        onPresentationScreen: (function () {
          try { return CS.detectPresentationScreen(); } catch (e) { return "error: " + e.message; }
        })(),
        onWrongAnswerScreen: (function () {
          try { return CS.detectWrongAnswerCorrection(); } catch (e) { return "error: " + e.message; }
        })(),
        onCorrectAnswerScreen: (function () {
          try { return CS.detectMultipleChoiceCorrect() || CS.detectTypedCorrect(); }
          catch (e) { return "error: " + e.message; }
        })(),
        deckIndexSize: this.deckIndex ? Object.keys(this.deckIndex).length : 0,
        sampleMediaUrl: (Array.isArray(this.currentExamples) && this.currentExamples.length)
          ? this.mediaUrl(this.currentExamples[0], "sound") : null,
        panelInDom: !!document.getElementById('context-sentences-panel'),
        inlineBoxInDom: !!document.getElementById('context-sentence-inline'),
      };
      console.log("=== Memrise Context Sentences diagnosis ===");
      console.log(JSON.stringify(info, null, 2));
      this.selfCheckVisibility();
      return info;
    },

    //+------------------------------------------------------
    //| SESSION / WORD WATCHING
    //+------------------------------------------------------

    currentCourseId: function () {
      var m = /[?&]course_id=(\d+)/.exec(window.location.href);
      if (m) { return Number(m[1]); }
      m = /\/course\/(\d+)/.exec(window.location.pathname);
      if (m) { return Number(m[1]); }
      try { return MEMRISE.garden.session_params.course_id; } catch (e) { /* ignore */ }
      return null;
    },

    watchSession: function () {
      var attempts = 0;
      var attempt = function () {
        attempts++;
        try {
          if (typeof MEMRISE !== "undefined" && MEMRISE.garden) {
            var session = MEMRISE.garden.session;
            if (session && typeof session.make_box === "function" && !session.make_box.__contextSentences) {
              this.hookMemrise();
            }
          } else if (attempts === 16 && this.mode === "waiting") {
            log("MEMRISE.garden not found after ~8s - switching to DOM fallback mode");
            this.startDomMode();
          }
        } catch (e) {
          warn("session watch error: " + e.message);
        }
      }.bind(this);
      attempt();
      setInterval(attempt, 500);
    },

    hookMemrise: function () {
      if (MEMRISE.garden.session.make_box.__contextSentences) { return; }
      this.cachedMakeBox = MEMRISE.garden.session.make_box;
      var make_box = this.make_box;
      make_box.__contextSentences = true;
      MEMRISE.garden.session.make_box = make_box;
      this.mode = "memrise";
      this.stopDomMode();
      log("hooked Memrise session");
      if (MEMRISE.garden.box) { this.set_content(MEMRISE.garden.box); }
    },

    make_box: function () {
      var context = CS.cachedMakeBox.apply(this, arguments);
      try { CS.set_content(context); } catch (e) { warn("set_content failed: " + e.message); }
      return context;
    },

    set_content: function (context) {
      if (!context || !context.learnable || !context.learnable.item || !context.learnable.definition) { return; }
      var itemVal = context.learnable.item.kind === "text" ? context.learnable.item.value : "";
      var defVal  = context.learnable.definition.kind === "text" ? context.learnable.definition.value : "";
      var word = containsJapanese(itemVal) ? itemVal : (containsJapanese(defVal) ? defVal : "");
      this.setCurrentWord(word);
    },

    startDomMode: function () {
      if (this.mode === "memrise" || this.domObserver) { return; }
      this.mode = "dom";
      var scheduled = null;
      this.domObserver = new MutationObserver(function () {
        if (scheduled) { return; }
        scheduled = setTimeout(function () { scheduled = null; CS.readWordFromDom(); }, 250);
      });
      this.domObserver.observe(document.body, { childList: true, subtree: true, characterData: true });
      this.readWordFromDom();
    },

    stopDomMode: function () {
      if (this.domObserver) { this.domObserver.disconnect(); this.domObserver = null; }
    },

    /**
     * DOM fallback: gather candidate text from likely prompt/label elements
     * and use the first one that actually contains Japanese characters.
     * Doesn't try to be clever about round type - if a screen shows the
     * Japanese word anywhere matching these selectors, this finds it.
     */
    readWordFromDom: function () {
      var candidates = [];

      var cards = document.querySelectorAll('[data-testid$="LearnableCard"]');
      for (var c = 0; c < cards.length; c++) {
        var headings = cards[c].querySelectorAll('h2, h3');
        for (var i = 0; i < headings.length; i++) {
          candidates.push((headings[i].textContent || "").trim());
        }
      }

      for (var s = 0; s < DEFAULT_PROMPT_SELECTORS.length; s++) {
        try {
          var els = document.querySelectorAll(DEFAULT_PROMPT_SELECTORS[s]);
          for (var j = 0; j < els.length; j++) {
            candidates.push((els[j].textContent || "").trim());
          }
        } catch (e) { /* bad selector, ignore */ }
      }

      var found = "";
      for (var k = 0; k < candidates.length; k++) {
        if (candidates[k] && candidates[k].length < 60 && containsJapanese(candidates[k])) {
          found = candidates[k];
          break;
        }
      }
      if (found) { this.setCurrentWord(found); }
    },

    setCurrentWord: function (word) {
      if (!word || word === this.currentWord) { return; }
      log("word changed -> \u201c" + word + "\u201d");
      this.currentWord = word;
      this.currentExamples = undefined;
      this.autoplayIndex = null; // fresh random pick for the new word
      // New word: clear the example injected under the previous one.
      this.removeInjectedExample();
      this.setStatus("Looking up \u201c" + word + "\u201d\u2026");
      this.refreshInlineBox();
      this.fetchExamples(word);
    },

    //+------------------------------------------------------
    //| IMMERSION KIT
    //+------------------------------------------------------

    /**
     * Resolve a deck slug (what the search endpoint puts in example.title,
     * e.g. "kanon__2006_") to its real metadata: the display title
     * ("Kanon (2006)") and category ("anime"). Both come from /index_meta.
     */
    deckMeta: function (slug) {
      if (!slug) { return null; }
      var meta = this.deckIndex && this.deckIndex[slug];
      if (meta && meta.title && meta.category) { return meta; }
      return null;
    },

    /** Human-readable source name, falling back to a de-slugged guess if /index_meta hasn't loaded or lacks this deck. */
    displayTitle: function (example) {
      var meta = this.deckMeta(example.title);
      if (meta) { return meta.title; }
      // Fallback: turn "your_lie_in_april" into "Your Lie In April" so it at
      // least reads as a title rather than a slug. Can't recover a
      // parenthesised year this way, hence /index_meta being preferred.
      return String(example.title || "").replace(/_+/g, " ").trim()
        .replace(/\b\w/g, function (c) { return c.toUpperCase(); });
    },

    /**
     * Build a media URL for an example.
     *
     * The v2 search endpoint returns a BARE FILENAME in example.image /
     * example.sound (e.g. "A_ClannadS01_E06....mp3") and NO category field
     * at all - example.title is only the deck slug. The previous version of
     * this script assumed full *_url fields plus an example.category and so
     * always resolved to null, which is why no audio buttons ever rendered.
     *
     * The CDN path needs the deck's PRETTY title and its category, both of
     * which only exist in /index_meta - so without that index loaded, there
     * is no way to construct a working URL (the slug form returns 403).
     */
    mediaUrl: function (example, kind) {
      if (!example) { return null; }
      var direct = example[kind + "_url"];
      if (direct) { return direct; } // honoured if the API ever starts sending absolute urls

      var file = example[kind];
      if (!file) { return null; }

      var meta = this.deckMeta(example.title);
      if (!meta) { return null; }

      return IMMERSIONKIT_MEDIA_BASE
        + encodeURIComponent(meta.category) + "/"
        + encodeURIComponent(meta.title) + "/media/"
        + encodeURIComponent(file);
    },

    pickExamples: function (data) {
      if (!data || !Array.isArray(data.examples) || data.examples.length === 0) { return []; }
      var limit = Math.max(1, Number(this.settings.exampleLimit) || 5);
      // Prefer ones with an actual sound file - a silent card doesn't serve
      // either the inline box or the "pause and let it play" popup well.
      var withSound = data.examples.filter(function (ex) { return CS.mediaUrl(ex, "sound"); });
      var pool = (withSound.length ? withSound : data.examples);
      return pool.slice(0, limit);
    },

    /**
     * Loads the deck index (slug -> {title, category, tags}) once. Every
     * media URL depends on it, so word lookups wait for it rather than
     * racing it - otherwise the first word or two of a session would render
     * with no audio button and a slug for a source name.
     */
    loadDeckIndex: function () {
      if (this.deckIndexPromise) { return this.deckIndexPromise; }

      var cached = storageGet(LOCALSTORAGE_DECKMETA, null);
      if (cached && cached.data && cached.fetchedAt
          && (Date.now() - cached.fetchedAt) < 7 * 24 * 60 * 60 * 1000) {
        this.deckIndex = cached.data;
        log("deck index loaded from cache (" + Object.keys(this.deckIndex).length + " decks)");
        this.deckIndexPromise = Promise.resolve(this.deckIndex);
        return this.deckIndexPromise;
      }

      log("fetching deck index (" + IMMERSIONKIT_INDEX_PATH + ")");
      this.deckIndexPromise = fetch(IMMERSIONKIT_ORIGIN + IMMERSIONKIT_INDEX_PATH)
        .then(function (res) {
          if (!res.ok) { throw new Error("HTTP " + res.status); }
          return res.json();
        })
        .then(function (json) {
          var data = json && json.data ? json.data : json;
          CS.deckIndex = data || {};
          storageSet(LOCALSTORAGE_DECKMETA, { data: CS.deckIndex, fetchedAt: Date.now() });
          log("deck index ready (" + Object.keys(CS.deckIndex).length + " decks)");
          return CS.deckIndex;
        })
        .catch(function (e) {
          warn("deck index fetch failed (" + e.message + ") - audio/images unavailable and "
            + "source names will fall back to de-slugged guesses");
          CS.deckIndex = CS.deckIndex || {};
          return CS.deckIndex;
        });
      return this.deckIndexPromise;
    },

    fetchExamples: function (word) {
      var cached = this.cache[word];
      if (cached) {
        this.currentExamples = cached.examples;
        this.renderInlineBox();
        return;
      }
      if (this.pendingInflight[word]) { return; }
      this.pendingInflight[word] = true;

      var url = IMMERSIONKIT_ORIGIN + IMMERSIONKIT_SEARCH_PATH + "?q=" + encodeURIComponent(word);
      var myToken = ++this.fetchToken;
      log("fetching examples for \u201c" + word + "\u201d");

      // Deck index first - pickExamples() filters on "has usable audio",
      // which can't be determined without it.
      this.loadDeckIndex().then(function () {
        return fetch(url);
      }).then(function (res) {
        if (!res.ok) { throw new Error("HTTP " + res.status); }
        return res.json();
      }).then(function (json) {
        delete CS.pendingInflight[word];
        var examples = CS.pickExamples(json);
        CS.cache[word] = { examples: examples, fetchedAt: Date.now() };
        storageSet(LOCALSTORAGE_CACHE, CS.cache);
        if (myToken !== CS.fetchToken) { return; } // a newer word has since taken over
        CS.currentExamples = examples;
        log("found " + examples.length + " example(s) for \u201c" + word + "\u201d");
        CS.renderInlineBox();
        CS.setStatus(examples.length ? "" : ("No example sentence found for \u201c" + word + "\u201d."));
      }).catch(function (e) {
        delete CS.pendingInflight[word];
        warn("Immersion Kit fetch failed: " + e.message);
        if (myToken === CS.fetchToken) { CS.setStatus("Lookup failed: " + e.message); }
      });
    },

    renderFurigana: function (sentenceWithFurigana, plainSentence) {
      var source = sentenceWithFurigana || plainSentence || "";
      // Immersion Kit uses "word[reading]" bracket notation for furigana.
      return source.replace(/([^\s\[\]]+)(?:\[([^\[\]]+)\])?/g, function (m, base, reading) {
        if (!base) { return m; }
        return reading
          ? "<ruby>" + escapeHtml(base) + "<rt>" + escapeHtml(reading) + "</rt></ruby>"
          : escapeHtml(base);
      });
    },

    buildSentenceCardContent: function (example, opts) {
      opts = opts || {};
      var wrapper = document.createElement('div');
      wrapper.className = 'context-sentence-content';

      var imageUrl = this.mediaUrl(example, "image");
      if (imageUrl) {
        var img = document.createElement('img');
        img.src = imageUrl;
        img.alt = "";
        img.loading = "lazy";
        img.className = 'context-sentence-image';
        wrapper.appendChild(img);
      }

      var textBox = document.createElement('div');
      textBox.className = 'context-sentence-text';

      var ja = document.createElement('div');
      ja.className = 'context-sentence-ja';
      ja.innerHTML = this.renderFurigana(example.sentence_with_furigana, example.sentence);
      textBox.appendChild(ja);

      var en = document.createElement('div');
      en.className = 'context-sentence-en';
      en.textContent = example.translation || "";
      if (opts.revealTranslationOnClick) {
        en.classList.add('context-sentence-hidden-until-click');
        en.addEventListener('click', function (e) {
          e.stopPropagation();
          en.classList.toggle('context-sentence-revealed');
        });
      }
      textBox.appendChild(en);

      if (example.title) {
        var src = document.createElement('div');
        src.className = 'context-sentence-source';
        src.textContent = this.displayTitle(example);
        textBox.appendChild(src);
      }

      wrapper.appendChild(textBox);
      return wrapper;
    },

    /**
     * Picks a random example, preferring ones that actually have a usable
     * audio clip - a purely random pick could land on a silent example and
     * there'd be nothing to play. Falls back to any example if none of them
     * have audio.
     */
    pickRandomExample: function (examples) {
      if (!examples || !examples.length) { return null; }
      var withSound = examples.filter(function (ex) { return CS.mediaUrl(ex, "sound"); });
      var pool = withSound.length ? withSound : examples;
      return pool[Math.floor(Math.random() * pool.length)];
    },

    /**
     * Index (within currentExamples) of the side-panel card that autoplays.
     * Chosen once per word and remembered, so that merely toggling a setting
     * - which re-renders the panel - doesn't keep jumping to a different
     * clip mid-word. Reset in setCurrentWord.
     */
    autoplayIndexForCurrentWord: function () {
      if (this.autoplayIndex !== null && this.autoplayIndex !== undefined) {
        return this.autoplayIndex;
      }
      var examples = this.currentExamples || [];
      if (!examples.length) { return 0; }

      var candidates = [];
      for (var i = 0; i < examples.length; i++) {
        if (this.mediaUrl(examples[i], "sound")) { candidates.push(i); }
      }
      if (!candidates.length) { for (var j = 0; j < examples.length; j++) { candidates.push(j); } }

      this.autoplayIndex = candidates[Math.floor(Math.random() * candidates.length)];
      return this.autoplayIndex;
    },
    buildInlineExampleCard: function (example, autoplay) {
      var card = document.createElement('div');
      card.className = 'context-sentence-card';
      card.appendChild(this.buildSentenceCardContent(example, { revealTranslationOnClick: true }));

      var soundUrl = this.mediaUrl(example, "sound");
      if (soundUrl) {
        var audio = new Audio(soundUrl);
        this.inlineAudioEls.push(audio);

        var playBtn = document.createElement('button');
        playBtn.type = 'button';
        playBtn.className = 'context-sentence-replay';
        playBtn.textContent = autoplay ? '\uD83D\uDD0A Playing\u2026' : '\u25B6 Play';
        playBtn.addEventListener('click', function (e) {
          e.stopPropagation();
          // Stop every other example's audio first, so clicking through
          // several in a row never stacks them on top of each other.
          CS.inlineAudioEls.forEach(function (other) {
            if (other !== audio) { try { other.pause(); } catch (err) { /* ignore */ } }
          });
          try { audio.currentTime = 0; } catch (err) { /* ignore */ }
          audio.play().catch(function (err) { warn("audio play failed: " + err.message); });
        });
        audio.addEventListener('play',  function () { playBtn.textContent = '\uD83D\uDD0A Playing\u2026'; });
        audio.addEventListener('ended', function () { playBtn.textContent = '\u25B6 Play'; });
        audio.addEventListener('pause', function () { playBtn.textContent = '\u25B6 Play'; });
        audio.addEventListener('error', function () {
          playBtn.textContent = '\u26A0 Audio unavailable';
          playBtn.disabled = true;
        });
        card.appendChild(playBtn);

        if (autoplay) {
          audio.play().catch(function () {
            // Autoplay blocked until the page has a user gesture - the
            // Play button above still works, so this fails silently.
            playBtn.textContent = '\u25B6 Play';
          });
        }
      } else {
        // Be explicit rather than silently omitting the control - if this
        // shows up for everything, the deck index failed to load (check the
        // console) rather than the clips genuinely not existing.
        var noAudio = document.createElement('div');
        noAudio.className = 'context-sentence-noaudio';
        noAudio.textContent = 'No audio for this example';
        card.appendChild(noAudio);
      }
      return card;
    },

    ensureInlinePanel: function () {
      var panel = document.getElementById('context-sentence-inline');
      if (panel) { return panel; }
      panel = document.createElement('div');
      panel.id = 'context-sentence-inline';
      rootContainer().appendChild(panel);
      return panel;
    },

    refreshInlineBox: function () {
      var panel = document.getElementById('context-sentence-inline');
      if (!this.settings.enabled || !this.settings.showInlineBox) {
        if (panel) { panel.style.display = 'none'; }
        return;
      }
      this.renderInlineBox();
    },

    renderInlineBox: function () {
      if (!this.settings.enabled || !this.settings.showInlineBox) { return; }
      var panel = this.ensureInlinePanel();
      panel.style.display = 'block';

      // Stop any audio from the previous word's cards before rebuilding.
      this.inlineAudioEls.forEach(function (a) { try { a.pause(); } catch (e) { /* ignore */ } });
      this.inlineAudioEls = [];

      if (this.currentExamples === undefined) {
        panel.innerHTML = '<div class="context-sentence-loading">Looking up examples\u2026</div>';
        return;
      }
      if (!this.currentExamples.length) {
        panel.innerHTML = '<div class="context-sentence-loading">No example sentences found.</div>';
        return;
      }

      panel.innerHTML = "";
      var header = document.createElement('div');
      header.className = 'context-sentence-header';
      header.textContent = this.currentWord + " \u2014 " + this.currentExamples.length + " example"
        + (this.currentExamples.length === 1 ? "" : "s");
      panel.appendChild(header);

      // Which card autoplays is random rather than always the first one.
      var autoIndex = this.autoplayIndexForCurrentWord();
      for (var i = 0; i < this.currentExamples.length; i++) {
        var autoplayThisOne = (i === autoIndex) && this.settings.autoplayFirstExample;
        panel.appendChild(this.buildInlineExampleCard(this.currentExamples[i], autoplayThisOne));
      }
    },

    //+------------------------------------------------------
    //| CORRECT-ANSWER DETECTION (heuristic - see header comment)
    //+------------------------------------------------------

    looksGreen: function (el) {
      if (!el) { return false; }
      var style;
      try { style = window.getComputedStyle(el); } catch (e) { return false; }
      var color = style.backgroundColor || style.borderColor || "";
      var m = /rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/.exec(color);
      if (!m) { return false; }
      var r = +m[1], g = +m[2], b = +m[3];
      return g > r + 20 && g > b + 20;
    },

    detectMultipleChoiceCorrect: function () {
      var card = document.querySelector('[data-testid="testLearnableCard"]');
      if (!card) { return false; }
      var buttons = card.querySelectorAll('button');
      for (var i = 0; i < buttons.length; i++) {
        if (this.looksGreen(buttons[i])) { return true; }
      }
      return false;
    },

    detectTypedCorrect: function () {
      var input = document.querySelector('[data-testid="typing-response-input"]');
      if (!input || !input.value) { return false; }
      return this.looksGreen(input) || this.looksGreen(input.parentElement);
    },

    /** Generic catch-all: a toast/label announcing correctness in text. */
    detectGenericCorrectToast: function (node) {
      if (!node || node.nodeType !== 1) { return false; }
      var text = (node.textContent || "").trim();
      if (text.length > 40) { return false; } // avoid matching large unrelated blocks
      return /^(correct|nice|great|perfect|well done|excellent|awesome)\b/i.test(text);
    },

    /**
     * The "new word" presentation screen (confirmed in the Audio Provider
     * script as testid presentationLearnableCard). The word is already fully
     * revealed here - there's no answer to spoil - so the example can show
     * immediately rather than waiting for anything.
     *
     * Matched generically against any *LearnableCard that ISN'T the
     * interactive test card, since Memrise reuses the same card component
     * across presentation-style screens.
     */
    detectPresentationScreen: function () {
      if (document.querySelector('[data-testid="presentationLearnableCard"]')) { return true; }
      var cards = document.querySelectorAll('[data-testid$="LearnableCard"]');
      for (var i = 0; i < cards.length; i++) {
        if (cards[i].getAttribute('data-testid') === 'testLearnableCard') { continue; }
        // A card showing both languages under their own <label>s is the
        // presentation layout - it has the answer on it already.
        if (cards[i].querySelector('label')) { return true; }
      }
      return false;
    },

    /**
     * The "you got it wrong, here's the correct answer" screen. Ported from
     * the Audio Provider script's readWrongAnswerCorrection: neither the
     * multiple-choice nor the typed-answer detector fires on a wrong answer
     * (one scans clickable buttons, the other only matches when YOUR input
     * turned green), yet the correct word is on screen and is exactly when
     * an example sentence is most useful.
     *
     * Two best-effort passes: a "correct" green highlight on a static (non
     * clicked) element inside a card, then a handful of testid patterns some
     * builds use for correction text.
     */
    detectWrongAnswerCorrection: function () {
      var cards = document.querySelectorAll('[data-testid$="LearnableCard"]');
      for (var c = 0; c < cards.length; c++) {
        var candidates = cards[c].querySelectorAll('button, li, [role="option"], [role="button"]');
        for (var i = 0; i < candidates.length; i++) {
          if (this.looksGreen(candidates[i])) {
            var t = (candidates[i].textContent || "").trim();
            if (t && t.length < 120) { return true; }
          }
        }
      }
      try {
        var sel = '[data-testid$="answer-text"], [data-testid$="correct-text"],'
          + ' [data-testid*="correction"], [data-testid*="correctAnswer"]';
        var els = document.querySelectorAll(sel);
        for (var j = 0; j < els.length; j++) {
          var txt = (els[j].textContent || "").trim();
          if (txt && txt.length < 120) { return true; }
        }
      } catch (e) { /* ignore */ }
      return false;
    },

    watchForCorrectAnswer: function () {
      // Note: there is deliberately no separate "already fired for this word"
      // variable here. injectedFor is the single source of truth, and it gets
      // reset by removeInjectedExample() - so once an example is cleared, the
      // same word can legitimately show one again later in the session. An
      // extra guard here used to block exactly that.
      var checkAndFire = function () {
        if (!CS.settings.enabled) { return; }

        // Clear a stale example from the previous question first, so moving
        // on always tidies up even when no new word gets recognised.
        try { CS.clearInjectedIfScreenMovedOn(); } catch (e) { warn("cleanup failed: " + e.message); }

        if (!CS.currentWord || CS.injectedFor === CS.currentWord) { return; }

        // Which screen are we on, and is showing enabled for it?
        var reason = null;
        if (CS.detectMultipleChoiceCorrect() || CS.detectTypedCorrect()) {
          if (!CS.settings.showOnCorrect) { return; }
          reason = "correct";
        } else if (CS.detectWrongAnswerCorrection()) {
          if (!CS.settings.showOnIncorrect) { return; }
          reason = "incorrect";
        } else if (CS.detectPresentationScreen()) {
          if (!CS.settings.showOnPresentation) { return; }
          reason = "presentation";
        }
        if (!reason) { return; }

        log("showing example (" + reason + " screen)");
        CS.showSentenceForCorrectAnswer();
      };

      // Poll as a backstop (covers round types with no DOM mutation to
      // hang an observer callback off of, e.g. a CSS-only highlight).
      setInterval(checkAndFire, 250);

      var observer = new MutationObserver(function (mutations) {
        checkAndFire();
        for (var i = 0; i < mutations.length; i++) {
          var added = mutations[i].addedNodes;
          for (var j = 0; j < added.length; j++) {
            if (CS.detectGenericCorrectToast(added[j])) {
              if (CS.settings.showOnCorrect && CS.currentWord
                  && CS.injectedFor !== CS.currentWord) {
                CS.showSentenceForCorrectAnswer();
              }
            }
          }
        }
      });
      observer.observe(document.body, { childList: true, subtree: true });
    },

    //+------------------------------------------------------
    //| IN-PAGE INJECTION (shown after a correct answer)
    //+------------------------------------------------------

    /**
     * Find the element to slot the example in after, so it appears
     * underneath the question/answer area rather than floating over it.
     * Walks a list of increasingly generic candidates because Memrise's
     * class names are build-hashed and can't be relied on; the last resort
     * is <body>, which just puts it at the end of the page.
     */
    findInjectionTarget: function () {
      var selectors = [
        '[data-testid="presentationLearnableCard"]',
        '[data-testid="testLearnableCard"]',
        '[data-testid$="LearnableCard"]',
        '[data-testid="typing-response-input"]',
        '[data-testid$="prompt-text"]'
      ];
      for (var i = 0; i < selectors.length; i++) {
        var el = document.querySelector(selectors[i]);
        if (!el) { continue; }
        // Climb to a reasonably wide block so the injected card lines up
        // with the exercise rather than being nested inside a narrow label.
        var node = el;
        for (var up = 0; up < 3 && node.parentElement; up++) {
          if (node.getBoundingClientRect().width > 300) { break; }
          node = node.parentElement;
        }
        return node;
      }
      return null;
    },

    removeInjectedExample: function () {
      var existing = document.getElementById('context-sentence-injected');
      if (existing) {
        if (existing.__audio) { try { existing.__audio.pause(); } catch (e) { /* ignore */ } }
        existing.remove();
      }
      this.injectedFor = null;
      this.injectedAnchor = null;
      this.injectedMisses = 0;
    },

    /**
     * Clears the injected example as soon as the session moves on.
     *
     * Tying removal only to "a new word was detected" (which is all the
     * previous version did) left the old example hanging around in two
     * cases: during the gap between screens before the next word is
     * recognised, and - permanently - when the next question happened to
     * use the SAME word, since the word never "changed".
     *
     * So this instead watches the screen itself. Two independent triggers:
     *   1. the element we anchored to has been detached from the document
     *      (React swapped the card out for the next question), or
     *   2. none of the reveal conditions hold any more, i.e. we're back on
     *      a plain unanswered question.
     *
     * (2) is debounced by a couple of polls, because the green highlight can
     * flicker off for a frame mid-transition and a single miss shouldn't rip
     * the example away while you're still reading it.
     */
    clearInjectedIfScreenMovedOn: function () {
      var block = document.getElementById('context-sentence-injected');
      if (!block) { return; }

      if (this.injectedAnchor && !document.contains(this.injectedAnchor)) {
        log("clearing example - the card it was attached to is gone");
        this.removeInjectedExample();
        return;
      }

      var stillRevealed = this.detectMultipleChoiceCorrect() || this.detectTypedCorrect()
        || this.detectWrongAnswerCorrection() || this.detectPresentationScreen();

      if (stillRevealed) { this.injectedMisses = 0; return; }

      this.injectedMisses = (this.injectedMisses || 0) + 1;
      if (this.injectedMisses >= 2) { // ~500ms at the 250ms poll interval
        log("clearing example - moved on to the next question");
        this.removeInjectedExample();
      }
    },

    /**
     * Injects the example directly into the page flow underneath the
     * exercise. No overlay, no click-blocking, no auto-continue - the
     * session carries on exactly as normal and this just sits below it
     * until the next word replaces it.
     */
    showSentenceForCorrectAnswer: function () {
      if (!this.currentExamples || !this.currentExamples.length) { return; }
      if (this.injectedFor === this.currentWord) { return; } // already showing for this word

      this.removeInjectedExample();

      // Random pick, not always the first - a different clip each time this
      // word comes round (and each time the example is re-shown).
      var example = this.pickRandomExample(this.currentExamples);
      if (!example) { return; }
      var target = this.findInjectionTarget();

      var block = document.createElement('div');
      block.id = 'context-sentence-injected';
      block.appendChild(this.buildSentenceCardContent(example, { revealTranslationOnClick: false }));

      var soundUrl = this.mediaUrl(example, "sound");
      if (soundUrl) {
        var audio = new Audio(soundUrl);
        block.__audio = audio;

        var playBtn = document.createElement('button');
        playBtn.type = 'button';
        playBtn.className = 'context-sentence-replay';
        playBtn.textContent = '\u25B6 Play';
        playBtn.addEventListener('click', function (e) {
          e.preventDefault();
          e.stopPropagation();
          try { audio.currentTime = 0; } catch (err) { /* ignore */ }
          audio.play().catch(function () {});
        });
        audio.addEventListener('play',  function () { playBtn.textContent = '\uD83D\uDD0A Playing\u2026'; });
        audio.addEventListener('ended', function () { playBtn.textContent = '\u25B6 Play'; });
        audio.addEventListener('pause', function () { playBtn.textContent = '\u25B6 Play'; });
        block.appendChild(playBtn);

        if (this.settings.autoplayOnCorrect) {
          // Stop the side-panel clips first so the two don't talk over
          // each other the moment an answer lands.
          this.inlineAudioEls.forEach(function (a) { try { a.pause(); } catch (e) { /* ignore */ } });
          audio.play().catch(function () { /* autoplay blocked - button still works */ });
        }
      }

      if (target && target.parentNode) {
        target.parentNode.insertBefore(block, target.nextSibling);
      } else {
        (document.body || rootContainer()).appendChild(block);
      }
      this.injectedFor = this.currentWord;
      // Remembered so the block can be cleared the moment this card is
      // swapped out for the next question - see clearInjectedIfScreenMovedOn.
      this.injectedAnchor = target || null;
      this.injectedMisses = 0;
    },

  };
```

```javascript
      if (Array.isArray(obj)) { obj.forEach(fn); }
      else { Object.keys(obj).forEach(function (k) { fn(obj[k], k); }); }
    },
    find: function (arr, fn) {
      if (!arr) { return undefined; }
      for (var i = 0; i < arr.length; i++) { if (arr[i] && fn(arr[i])) { return arr[i]; } }
      return undefined;
    },
    remove: function (arr, fn) {
      for (var i = arr.length - 1; i >= 0; i--) { if (fn(arr[i])) { arr.splice(i, 1); } }
    },
    indexBy: function (arr, key) {
      var out = {};
      (arr || []).forEach(function (x) { if (x && x[key] != null) { out[x[key]] = x; } });
      return out;
    }
  };

    /**
     * "Override all audio" means: only OUR generated audio is ever heard.
     *
     * Patching HTMLMediaElement.prototype.play (which is all this used to do)
     * catches only one of the several ways a web app can make noise, which is
     * why Memrise kept talking over us. All of these bypass play() entirely:
     *
     *   - <audio autoplay> inserted into the DOM. The browser starts it
     *     itself; no script ever calls play(), so no patch can intercept it.
     *   - the Web Audio API (AudioContext + AudioBufferSourceNode.start()),
     *     which is what most apps use for short, low-latency clips like
     *     pronunciation - it never touches a media element at all.
     *   - an element that began playing before we were even installed.
     *
     * So the silencer is layered, and every layer is reversible:
     *   1. play() is blocked for anything we didn't create.
     *   2. a capture-phase 'play'/'playing' listener on document catches
     *      anything that started by any other means, the moment it starts.
     *      (Media events don't bubble, but a capturing listener on document
     *      still sees them, which makes this a reliable catch-all.)
     *   3. every non-owned <audio>/<video> is muted and pinned to volume 0,
     *      so even audio we couldn't stop is inaudible.
     *   4. Web Audio source nodes are stopped from starting.
     *   5. a MutationObserver silences new elements as they're inserted, plus
     *      a slow sweep for anything that slipped through.
     *
     * Turning override off restores every element's original muted/volume/
     * autoplay values, so Memrise's own audio works normally again.
     */
    installAudioSilencer: function () {
      this.silencedElements = [];
      this.blockedAudioCount = 0;

      // --- 1. play() ---
      try {
        if (window.HTMLMediaElement && !HTMLMediaElement.prototype.play.__audioProviderPatched) {
          var nativePlay = HTMLMediaElement.prototype.play;
          var patchedPlay = function () {
            if (AudioProvider.overrideActive() && !this.__audioProviderOwned) {
              AudioProvider.noteBlockedAudio("play() call");
              AudioProvider.silenceMediaElement(this);
              // Callers often .then()/.catch() this - resolve rather than throw.
              return Promise.resolve();
            }
            return nativePlay.apply(this, arguments);
          };
          patchedPlay.__audioProviderPatched = true;
          HTMLMediaElement.prototype.play = patchedPlay;
        }
      } catch (e) {
        log("could not patch media play(): " + e.message);
      }

      // --- 2. anything that started without calling play() ---
      try {
        var onStart = function (e) {
          var el = e.target;
          if (!el || !AudioProvider.overrideActive()) { return; }
          if (el.__audioProviderOwned) { return; }
          AudioProvider.noteBlockedAudio("autoplay/" + e.type);
          AudioProvider.silenceMediaElement(el);
        };
        document.addEventListener('play', onStart, true);
        document.addEventListener('playing', onStart, true);
        document.addEventListener('loadeddata', onStart, true);
      } catch (e) {
        log("could not attach media event guard: " + e.message);
      }

      // --- 4. Web Audio ---
      this.installWebAudioGuard();

      // --- 5. new elements, plus a periodic sweep ---
      try {
        var scheduled = null;
        this.mediaObserver = new MutationObserver(function () {
          if (scheduled) { return; }
          scheduled = setTimeout(function () {
            scheduled = null;
            AudioProvider.sweepMediaElements();
          }, 0);
        });
        this.mediaObserver.observe(document.documentElement || document.body,
          { childList: true, subtree: true });
      } catch (e) {
        log("could not observe for new media elements: " + e.message);
      }

      this.mediaSweeper = setInterval(function () {
        AudioProvider.sweepMediaElements();
      }, 1000);

      this.sweepMediaElements();
    },

    /**
     * Web Audio never goes near a media element, so none of the above would
     * see it. Blocking the source node's start() stops the sound at the only
     * point every Web Audio playback has to pass through. We generate our own
     * audio with plain <audio> elements and never use Web Audio, so there's
     * nothing of ours to accidentally catch here.
     */
    installWebAudioGuard: function () {
      var targets = [];
      try {
        if (window.AudioBufferSourceNode) { targets.push(window.AudioBufferSourceNode.prototype); }
        if (window.AudioScheduledSourceNode) { targets.push(window.AudioScheduledSourceNode.prototype); }
        if (window.OscillatorNode) { targets.push(window.OscillatorNode.prototype); }
      } catch (e) { /* ignore */ }

      targets.forEach(function (proto) {
        try {
          if (!proto || !proto.start || proto.start.__audioProviderPatched) { return; }
          var nativeStart = proto.start;
          var patched = function () {
            if (AudioProvider.overrideActive()) {
              AudioProvider.noteBlockedAudio("Web Audio source");
              return undefined;
            }
            return nativeStart.apply(this, arguments);
          };
          patched.__audioProviderPatched = true;
          proto.start = patched;
        } catch (e) { /* ignore */ }
      });
    },

    noteBlockedAudio: function (how) {
      this.blockedAudioCount = (this.blockedAudioCount || 0) + 1;
      // Only log the first few, or a looping element would flood the console.
      if (this.blockedAudioCount <= 5) {
        log("override mode: silenced Memrise audio (" + how + ")");
      }
    },

    /**
     * Mutes, pauses and de-autoplays one element, remembering what it was so
     * it can be handed back untouched when override is switched off.
     */
    silenceMediaElement: function (el) {
      if (!el || el.__audioProviderOwned) { return; }

      if (!el.__audioProviderSilenced) {
        el.__audioProviderSilenced = {
          muted: el.muted, volume: el.volume, autoplay: el.autoplay
        };
        this.silencedElements.push(el);
      }
      try { el.autoplay = false; } catch (e) { /* read-only in some builds */ }
      try { el.muted = true; } catch (e) { /* ignore */ }
      try { el.volume = 0; } catch (e) { /* ignore */ }
      try { if (!el.paused) { el.pause(); } } catch (e) { /* ignore */ }
    },

    restoreMediaElement: function (el) {
      var previous = el && el.__audioProviderSilenced;
      if (!previous) { return; }
      try { el.muted = previous.muted; } catch (e) { /* ignore */ }
      try { el.volume = previous.volume; } catch (e) { /* ignore */ }
      try { el.autoplay = previous.autoplay; } catch (e) { /* ignore */ }
      delete el.__audioProviderSilenced;
    },

    /**
     * Silences every media element on the page while override is on, and
     * puts them all back when it isn't. Called on DOM changes, on a timer,
     * and whenever the audio-mode setting changes.
     */
    sweepMediaElements: function () {
      var active = this.overrideActive();

      if (!active) {
        var previously = this.silencedElements || [];
        for (var j = 0; j < previously.length; j++) {
          this.restoreMediaElement(previously[j]);
        }
        this.silencedElements = [];
        return;
      }

      var els;
      try { els = document.querySelectorAll('audio, video'); }
      catch (e) { return; }

      for (var i = 0; i < els.length; i++) {
        this.silenceMediaElement(els[i]);
      }
    },

    resolveLanguage: function (label) {
      if (!label) { return null; }
      if (SPEECHSYNTHESIS_LANG[label] || GOOGLETTS_LANG[label] || VOICERSS_LANG[label]) {
        return label;
      }
      if (LANGUAGE_ALIASES[label]) { return LANGUAGE_ALIASES[label]; }

      var match = null;
      ALL_LANGUAGES.forEach(function (known) {
        var bare = known.replace(/\s*\(.*\)$/, "");
        if (label.toLowerCase().indexOf(bare.toLowerCase()) === 0
            && (!match || known.length > match.length)) {
          match = known;
        }
      });
      return match;
    },

    detectLanguage: function () {
      var found = null;
      var push = function (fn) {
        if (found) { return; }
        try {
          var v = fn();
          if (typeof v === "string" && v.length) { found = v; }
        } catch (e) { /* ignore */ }
      };
      push(function () { return MEMRISE.garden.session.category.name; });
      push(function () { return MEMRISE.garden.session_data.category.name; });
      push(function () { return MEMRISE.garden.session_data.course.category.name; });
      push(function () { return MEMRISE.garden.course.category.name; });
      push(function () { return MEMRISE.course.category.name; });
      return found;
    },

    detectSourceLanguage: function () {
      // Simple, predictable default: assume English until the user says
      // otherwise. The old browser-locale guess was unreliable (it guessed
      // wrong for plenty of people), so it's gone - manual override in the
      // dropdown is the only thing that changes this now, and only that
      // override gets remembered.
      return "English";
    },

    /**
     * The real, reliable source: every community course's own page has a
     * breadcrumb like Courses > Languages > European > Dutch, where the last
     * link is the target language and the first link's href encodes the
     * source language (/community/courses/english/...). This is what the
     * course itself declares, so it's far more accurate than guessing from
     * on-screen text or the browser's locale.
     */
    parseBreadcrumb: function (html) {
      try {
        var doc = new DOMParser().parseFromString(html, "text/html");
        var block = doc.querySelector('.course-breadcrumb');
        if (!block) { return null; }

        var links = Array.prototype.slice.call(block.querySelectorAll('a'));
        if (!links.length) { return null; }

        var targetText = (links[links.length - 1].textContent || "").trim();

        var srcHref = links[0].getAttribute('href') || "";
        var srcMatch = /\/community\/courses\/([a-z0-9-]+)\//i.exec(srcHref);
        var sourceSlug = srcMatch ? srcMatch[1] : null;
        var sourceText = sourceSlug
          ? sourceSlug.replace(/-/g, ' ').replace(/\b\w/g, function (c) { return c.toUpperCase(); })
          : null;

        if (!targetText) { return null; }
        return {
          target: this.resolveLanguage(targetText) || targetText,
          source: sourceText ? (this.resolveLanguage(sourceText) || sourceText) : null
        };
      } catch (e) {
        return null;
      }
    },

    /**
     * Fetches the course's own page (same origin, so a plain page fetch()
     * works - no GM bridge needed) and caches the result in localStorage
     * indefinitely, since a course's languages never change. Re-applies
     * language once resolved, so the very first word or two on a fresh course
     * may still use the rough guess until this completes.
     */
    fetchBreadcrumbLanguages: function (courseId) {
      if (!courseId) { return; }
      if (this.courseLangCache[courseId]) { return; }
      if (this.breadcrumbFetchInFlight === courseId) { return; }
      this.breadcrumbFetchInFlight = courseId;

      var url = "https://community-courses.memrise.com/community/course/" + courseId + "/";
      log("looking up course languages from " + url);

      fetch(url, { credentials: "same-origin" })
        .then(function (res) {
          if (!res.ok) { throw new Error("HTTP " + res.status); }
          return res.text();
        })
        .then(function (html) {
          AudioProvider.breadcrumbFetchInFlight = null;
          var parsed = AudioProvider.parseBreadcrumb(html);
          if (!parsed) {
            log("course-breadcrumb not found on course page - falling back to guesses");
            return;
          }
          AudioProvider.courseLangCache[courseId] = parsed;
          storageSet(LOCALSTORAGE_BREADCRUMB, JSON.stringify(AudioProvider.courseLangCache));
          log("course languages detected: target=" + parsed.target + ", source=" + parsed.source);
          AudioProvider.applyLanguage();
          if (AudioProvider.mode === "dom") { AudioProvider.readPromptFromDom(true); }
        })
        .catch(function (e) {
          AudioProvider.breadcrumbFetchInFlight = null;
          log("course-breadcrumb fetch failed: " + e.message);
        });
    },

    applyLanguage: function () {
      var courseId  = this.currentCourseId();
      var breadcrumb = this.courseLangCache[courseId] || null;

      this.detectedLanguage = breadcrumb ? breadcrumb.target : this.detectLanguage();

      var override = this.savedLanguages[courseId];
      var chosen   = (override && override !== AUTO) ? override : this.detectedLanguage;

      this.language            = this.resolveLanguage(chosen);
      this.lastError           = null;
      this.canFish             = !!this.fishKey;
      this.canGoogleTts        = true;
      this.canVoiceRss         = !!this.voiceRssKey;

      var select = document.getElementById('audio-provider-language');
      if (select) {
        select.value = (override && override !== AUTO) ? override : AUTO;
        var autoOption = select.querySelector('option[value="' + AUTO + '"]');
        if (autoOption) {
          var targetLabel = breadcrumb ? (breadcrumb.target + ", from course page")
            : this.detectedLanguage ? this.detectedLanguage
            : (this.breadcrumbFetchInFlight === courseId ? "looking up…" : "not detected");
          autoOption.textContent = AUTO + " (" + targetLabel + ")";
        }
      }

      // Source (the user's own) language. Defaults to English; the course
      // breadcrumb's source guess is shown for reference only (some courses
      // aren't aimed at English speakers), but the actual default stays
      // English until you pick something else in the dropdown - and that
      // choice is the only thing that gets remembered.
      this.detectedSourceLanguage = this.detectSourceLanguage();
      var srcOverride  = this.savedSourceLanguages[courseId];
      var srcChosen    = (srcOverride && srcOverride !== AUTO) ? srcOverride : this.detectedSourceLanguage;
      this.sourceLanguage = this.resolveLanguage(srcChosen) || srcChosen;

      var srcSelect = document.getElementById('audio-provider-sourcelanguage');
      if (srcSelect) {
        srcSelect.value = (srcOverride && srcOverride !== AUTO) ? srcOverride : AUTO;
        var srcAutoOption = srcSelect.querySelector('option[value="' + AUTO + '"]');
        if (srcAutoOption) {
          var sourceLabel = this.detectedSourceLanguage + ", default"
            + (breadcrumb && breadcrumb.source && breadcrumb.source !== this.detectedSourceLanguage
              ? " - course page suggests " + breadcrumb.source : "");
          srcAutoOption.textContent = AUTO + " (" + sourceLabel + ")";
        }
      }

      // Kick off (or rely on the cache for) the real detection. Only needed
      // in DOM mode - MEMRISE-hooked mode already has real category data.
      if (this.mode !== "memrise" && courseId
          && !(override && override !== AUTO) && !breadcrumb) {
        this.fetchBreadcrumbLanguages(courseId);
      }

      this.applyCoursePreset(courseId);
      // Fish voices are per language, so the field has to follow the target
      // language rather than the course.
      this.refreshFishVoiceField();
      this.refreshStatus();
    },

    /**
     * Heuristic only: checks whether `text` contains common function words
     * from the resolved source language. Used to avoid reading your own
     * language out loud when only the target language should be spoken.
     * Not real language detection - short prompts or cognates shared between
     * languages can occasionally fool it either way.
     */
    /**
     * How many distinct function words of `lang` appear in `text`.
     *
     * A count, not a yes/no, because the lists genuinely overlap - "is" is
     * both English and Dutch, "no" is both English and Spanish. Asking each
     * language "do you contain any of these?" made both answer yes and the
     * comparison came out as "no idea", which is exactly how source-language
     * text ended up being read aloud. Comparing counts settles it: "the dog
     * is big" scores 2 for English and 1 for Dutch.
     */
    stopwordScore: function (text, lang) {
      var words = STOPWORDS_BY_LANG[lang];
      if (!words || !text) { return 0; }

      var tokens = String(text).toLowerCase()
        .split(/[^a-zàâäéèêëïîôöùûüÿçñãõ]+/i)
        .filter(Boolean);

      var seen = {}, score = 0;
      for (var i = 0; i < tokens.length; i++) {
        var t = tokens[i];
        if (!seen[t] && words.indexOf(t) !== -1) { seen[t] = true; score++; }
      }
      return score;
    },

    looksLikeLanguage: function (text, lang) {
      return this.stopwordScore(text, lang) > 0;
    },

    looksLikeSourceLanguage: function (text) {
      return this.looksLikeLanguage(text, this.sourceLanguage);
    },

    /**
     * Hard evidence from the writing system.
     *   true  - this text is written in `lang`'s script
     *   false - this text definitely ISN'T (lang uses a non-Latin script and
     *           the text is Latin, or vice versa)
     *   null  - no evidence either way (both languages are Latin-script)
     */
    scriptEvidence: function (lang, text) {
      if (!lang || !text) { return null; }
      var scripts = LANGUAGE_SCRIPTS[lang];
      var t = String(text);

      if (!scripts) {
        // Latin-script language. Text in a non-Latin script definitely isn't it.
        var foreign = Object.keys(SCRIPT_PATTERNS).some(function (name) {
          return name !== "latin" && SCRIPT_PATTERNS[name].test(t);
        });
        return foreign ? false : null;
      }

      for (var i = 0; i < scripts.length; i++) {
        var p = SCRIPT_PATTERNS[scripts[i]];
        if (p && p.test(t)) { return true; }
      }
      // The language needs a script this text doesn't contain at all.
      return SCRIPT_PATTERNS.latin.test(t) ? false : null;
    },

    /**
     * Which side of the card a piece of text belongs to: "target", "source",
     * or null when there genuinely isn't enough evidence to say.
     *
     * Script beats stopwords, and stopwords are now checked against BOTH
     * languages instead of only the source. Checking one side was the old
     * bug: with a Dutch course, "de" hit the Dutch list and the English list
     * was never consulted, so which language won depended on the accident of
     * which list happened to contain the word.
     */
    classifyText: function (text) {
      if (!text) { return null; }
      var t = String(text).trim();
      if (!t) { return null; }
      if (this.language && this.language === this.sourceLanguage) { return "target"; }

      var targetScript = this.scriptEvidence(this.language, t);
      var sourceScript = this.scriptEvidence(this.sourceLanguage, t);

      if (targetScript === true && sourceScript !== true)  { return "target"; }
      if (sourceScript === true && targetScript !== true)  { return "source"; }
      if (targetScript === false && sourceScript !== false) { return "source"; }
      if (sourceScript === false && targetScript !== false) { return "target"; }

      var targetScore = this.stopwordScore(t, this.language);
      var sourceScore = this.stopwordScore(t, this.sourceLanguage);
      if (targetScore > sourceScore) { return "target"; }
      if (sourceScore > targetScore) { return "source"; }

      return null;
    },

    otherSide: function (side) {
      return side === "target" ? "source" : (side === "source" ? "target" : null);
    },

    /**
     * Maps a language NAME found on screen (e.g. a "Dutch" label) onto the
     * side of the card it represents.
     */
    sideOfLanguage: function (langName) {
      var resolved = this.resolveLanguage(langName) || langName;
      if (!resolved) { return null; }
      if (this.language && resolved === this.language) { return "target"; }
      if (this.sourceLanguage && resolved === this.sourceLanguage) { return "source"; }
      return null;
    },

    langForSide: function (side) {
      return side === "source" ? this.sourceLanguage : this.language;
    },

    /**
     * Whether the source language may be spoken at all. Two independent
     * opt-ins: ticking "Also play source language", or untickng "Only speak
     * target language". With the defaults (target-only ON, play-source OFF)
     * the answer is no, so your own language is never read aloud.
     */
    mayPlaySourceLanguage: function () {
      return !!this.playSourceLanguage || !this.targetOnly;
    },

    /**
     * Looks for a language name displayed near `el` (Memrise labels each
     * side of most screens with its language). Walks up a few ancestors and
     * stops as soon as a level names exactly one language - if a level names
     * two (i.e. it contains both sides of the card), that's ambiguous and
     * gets rejected rather than guessed.
     */
    findLanguageLabelNear: function (el) {
      var node = el, depth = 0;
      while (node && depth < 6) {
        var found = {};
        var candidates = node.querySelectorAll
          ? node.querySelectorAll('label, div[dir="auto"], span, h4, h5')
          : [];
        for (var i = 0; i < candidates.length; i++) {
          var t = (candidates[i].textContent || "").trim();
          if (!t || t.length > 30) { continue; }
          var resolved = ALL_LANGUAGES.indexOf(t) !== -1 ? t : null;
          if (!resolved && LANGUAGE_ALIASES[t]) { resolved = LANGUAGE_ALIASES[t]; }
          if (resolved) { found[resolved] = true; }
        }
        var names = Object.keys(found);
        if (names.length === 1) { return names[0]; }
        if (names.length > 1) { return null; }
        node = node.parentElement;
        depth++;
      }
      return null;
    },

    set_content: function (context) {
      if (!context || context.template === "end_of_session"
          || context.template === "speed-count-down") { return; }
      if (!context.learnable || !context.learnable.item || !context.learnable.definition) { return; }

      var courseId = context.course_id || this.sessionCourseId() || this.urlCourseId();

      if (this.courseId !== courseId) {
        this.courseId   = courseId;
        this.wordColumn = this.savedChoices[courseId]
          || (context.learnable.item.kind === "text" ? context.learnable.item.label
              : (context.learnable.definition.kind === "text" ? context.learnable.definition.label
                : NONE));
        this.applyLanguage();
        this.editAudioOptions(context.learnable);
      }

      var media = context.presentationData || context.testData || {};
      this.currentItem = {
        item:       context.learnable.item,
        definition: context.learnable.definition,
        audio:      media.audio ? media.audio.value : null
      };
      this.updateCurrentWord();

      // Speed sessions never call the audio player, so play it ourselves.
      // Source-language words are already handled (and gated) inside
      // updateCurrentWord, so only the target side is played here - without
      // this check an opted-in source word would be spoken twice.
      if (this.autoplay && this.currentWord && !this.skippedAsSource
          && this.currentWordLang === this.language) {
        var key = String(context.learnable.learnable_id) + "|" + context.template + "|" + this.currentWord;
        if (key !== this.lastPlayed) {
          this.lastPlayed = key;
          setTimeout(function () {
            try {
              AudioProvider.playGeneratedAudio(AudioProvider.currentWord, {},
                AudioProvider.currentWordLang);
            } catch (e) { /* page may be gone */ }
          }, 150);
        }
      }
    },

    editAudioOptions: function (word) {
      var html = `<option value="${NONE}">${NONE}</option>`;
      if (word.definition.kind === "text") {
        html += `<option value="${word.definition.label}">${word.definition.label}</option>`;
      }
      if (word.item.kind === "text") {
        html += `<option value="${word.item.label}">${word.item.label}</option>`;
      }
      var options = document.getElementById('audio-provider-options');
      if (!options) { return; }
      options.innerHTML = html;
      options.value     = this.wordColumn;
    },

    updateCurrentWord: function () {
      var item = this.currentItem;
      if (!item) { return; }

      var audio       = U.isArray(item.audio) ? item.audio[0] : item.audio,
          // "all" and "override" both mean "always generate ours"; only
          // "missing" (the default) restricts it to items Memrise itself
          // has no real audio for. The difference between "all" and
          // "override" is whether Memrise's own audio also gets silenced -
          // handled separately in fixMediaUrl().
          handleAudio = this.audioMode === AUDIO_MODE_ALL || this.audioMode === AUDIO_MODE_OVERRIDE
            || !audio || (audio && audio.normal === "AUDIO_PROVIDER");

      if (!handleAudio || this.wordColumn === NONE) {
        this.currentWord = "";
        this.skippedAsSource = false;
        this.answerPending = false;
        this.refreshStatus();
        this.updatePlayAgainButtons();
        return;
      }

      var column = U.find([item.definition, item.item], function (x) {
        return x && x.label === this.wordColumn;
      }.bind(this));

      this.currentWord = (column && column.value) || "";

      // "Only speak target language" used to only be enforced in DOM
      // fallback mode - the MEMRISE-hooked path below just spoke whatever
      // column was selected, with no check at all, so English kept getting
      // read even with the box ticked. A column's *label* is itself often
      // a language name (that's what populates the "Column to use"
      // dropdown), so prefer resolving that directly; fall back to the
      // stopword heuristic only if the label isn't a recognised language.
      // Work out which language this column actually holds. The column's
      // label is usually a language name (that's what fills the dropdown),
      // which is definitive; otherwise fall back to script/stopword
      // classification. The old code only ever asked "is this the source
      // language?" and, if not, spoke it with the TARGET voice - so a column
      // it couldn't identify got read in the wrong accent every time.
      var columnLanguage = column && this.resolveLanguage(column.label);
      var side = columnLanguage ? this.sideOfLanguage(columnLanguage) : null;
      if (!side && this.currentWord) { side = this.classifyText(this.currentWord); }
      if (!side) { side = "target"; }

      this.currentWordLang = this.langForSide(side);
      this.skippedAsSource = (side === "source") && !this.mayPlaySourceLanguage();
      this.answerPending = false;
      this.refreshStatus();
      this.updatePlayAgainButtons();

      if (this.skippedAsSource) { return; }

      if (side === "source" && this.autoplay && this.currentWord) {
        // Opted in, so read it - but with the SOURCE voice, never the target one.
        setTimeout(function () {
          try { AudioProvider.playGeneratedAudio(AudioProvider.currentWord, {}, AudioProvider.sourceLanguage); }
          catch (e) { /* page may be gone */ }
        }, 150);
        return;
      }

      // Warm the cache so playback isn't delayed by the round trip. Only when
      // autoplay is on - Fish Audio charges per request, so never prefetch a
      // word the user might not ask to hear. Prefetch through the same
      // language-aware engine order that playback will use, or the warmed
      // entry would be from the wrong engine and get thrown away.
      if (this.autoplay && this.currentWord) {
        this.prefetch(this.currentWord, this.currentWordLang);
      }
    },

    /**
     * Warms the cache using whichever engine playback would actually pick
     * for this language.
     */
    prefetch: function (word, lang) {
      lang = lang || this.language;
      var order = this.engineOrderFor(lang);
      if (!order.length) { return; }
      if (order[0] === "fish")   { this.fetchFishAudio(word, false, {}, lang); }
      else if (order[0] === "google") { this.fetchGoogleTts(word, false, {}, lang); }
    },

    watchSession: function () {
      var attempts = 0;
      var attempt = function () {
        attempts++;
        try {
          if (typeof MEMRISE !== "undefined" && MEMRISE.garden) {
            var session = MEMRISE.garden.session;
            if (session && typeof session.make_box === "function" && !session.make_box.__audioProvider) {
              this.override_memrise();
            }
          } else if (attempts === 16 && this.mode === "waiting") {
            // ~8s with no MEMRISE.garden: this build doesn't expose it.
            log("MEMRISE.garden not found - switching to DOM fallback mode");
            this.startDomMode();
          }
        } catch (e) {
          log("session watch error: " + e.message);
        }
      }.bind(this);

      attempt();
      this.sessionWatcher = setInterval(attempt, 500);
    },

    override_memrise: function () {
      if (MEMRISE.garden.session.make_box.__audioProvider) { return; }

      this.cached = this.cached || {};
      this.cached.make_box = MEMRISE.garden.session.make_box;

      var make_box = this.make_box;
      make_box.__audioProvider = true;
      MEMRISE.garden.session.make_box = make_box;

      if (MEMRISE.renderer && !MEMRISE.renderer.fixMediaUrl.__audioProvider) {
        this.cached.fixMediaUrl = MEMRISE.renderer.fixMediaUrl;
        this.fixMediaUrl.__audioProvider = true;
        MEMRISE.renderer.fixMediaUrl = this.fixMediaUrl;
      }
      if (MEMRISE.audioPlayer && !MEMRISE.audioPlayer.play.__audioProvider) {
        this.cached.play = MEMRISE.audioPlayer.play;
        this.play.__audioProvider = true;
        MEMRISE.audioPlayer.play = this.play;
      }

      this.mode     = "memrise";
      this.stopDomMode();
      this.courseId = null; // re-read column/language for the new session
      this.applyLanguage();
      this.populateScreenAudios();

      log("hooked session (" + (this.language || "language not detected") + ")");

      if (MEMRISE.garden.box) {
        this.set_content(MEMRISE.garden.box);
        this.ensurePlayerElement();
      }
    },

        }
      }
    },

    populateScreenAudios: function () {
      var garden = MEMRISE.garden;
      var learnables = garden.learnables
        || (garden.session_data && U.indexBy(garden.session_data.learnables, 'learnable_id'))
        || {};

      U.each(learnables, function (v, k) {
        var screenSets = [
          garden.screens ? garden.screens[k] : null,
          garden.session_data && garden.session_data.screens ? garden.session_data.screens[k] : null,
          garden.screen_template_map ? garden.screen_template_map[k] : null
        ];

        U.each(screenSets, function (screens) {
          U.each(screens, function (s) {
            s = U.isArray(s) ? s[0] : s;
            if (!s || typeof s !== "object") { return; }

            var hasAudio = s.audio && s.audio.value && s.audio.value.length;
            if (!hasAudio) {
              s.audio = {
                alternatives: [], direction: "target", kind: "audio",
                label: "Audio", style: [],
                value: [{ normal: "AUDIO_PROVIDER", slow: "AUDIO_PROVIDER" }]
              };
            }
          });
        });
      });
    },

    make_box: function () {
      var context = AudioProvider.cached.make_box.apply(this, arguments);

        AudioProvider.set_content(context);

      }

    },

    /**
     * Memrise runs image URLs through fixMediaUrl too, so blanking
     * everything in override mode (what this used to do) silenced the audio
     * but also broke the pictures on picture cards. Only audio is blanked
     * now; anything fixMediaUrl can't confidently identify as audio is left
     * alone and caught later by the element/Web Audio guards instead.
     */
    fixMediaUrl: function () {
      var arg   = arguments[0];
      var first = U.isArray(arg) ? arg[0] : arg;

      if (first === "AUDIO_PROVIDER") { return ""; }

      if (AudioProvider.overrideActive() && typeof first === "string"
          && /\.(mp3|ogg|oga|wav|m4a|aac|opus|weba|flac)(\?|#|$)/i.test(first)) {
        AudioProvider.noteBlockedAudio("fixMediaUrl");
        return "";
      }

      return AudioProvider.cached.fixMediaUrl.apply(this, arguments);
    },

    play: function () {
      if (document.body.classList.contains("audio-muted")) { return; }

      var shouldGenerate = !arguments[0] || arguments[0].url === "" || arguments[0].url == null;
      if (shouldGenerate) {
        if (AudioProvider.skippedAsSource) { return; }
        AudioProvider.playGeneratedAudio(AudioProvider.currentWord, {},
          AudioProvider.currentWordLang || AudioProvider.language);
      } else {
        return AudioProvider.cached.play.apply(this, arguments);
      }
    },

    /**
     * Some screens (confirmed: the "new word" presentation screen, testid
     * presentationLearnableCard) show BOTH languages side by side, each
     * under its own <label> naming the language (e.g. "Dutch" / "English").
     * That's unambiguous - no heuristic needed, just match the label text
     * against the resolved target/source language. Matched generically
     * against any *LearnableCard testid, since Memrise likely reuses the same
     * card component for the "you got it wrong, here's the answer" screen
     * too; if a card has no <label> pairs (e.g. the multiple-choice test
     * screen), this simply finds nothing and the caller falls back to the
     * plain prompt-text selector.
     *
     * Returns { target, source } - either may be null if that language's
     * label wasn't found on screen (never guessed, only ever a confirmed
     * label match).
     */
    readCardWithLanguageLabels: function () {
      var cards = document.querySelectorAll('[data-testid$="LearnableCard"]');
      for (var c = 0; c < cards.length; c++) {
        var labels = cards[c].querySelectorAll('label');
        if (!labels.length) { continue; }

        var pairs = [];
        for (var i = 0; i < labels.length; i++) {
          var label     = labels[i];
          var container = label.parentElement;
          var heading   = container ? container.querySelector('h2, h3') : null;
          if (!heading) { continue; }

          var langName = (label.textContent || "").trim();
          var text     = (heading.textContent || "").trim();
          if (text) { pairs.push({ lang: this.resolveLanguage(langName) || langName, text: text }); }
        }
        if (!pairs.length) { continue; }

        var targetPair = U.find(pairs, function (p) { return p.lang === this.language; }.bind(this));
        var sourcePair = U.find(pairs, function (p) { return p.lang === this.sourceLanguage; }.bind(this));
        if (targetPair || sourcePair) {
          return { target: targetPair ? targetPair.text : null, source: sourcePair ? sourcePair.text : null };
        }
      }
      return null;
    },

    /**
     * On the multiple-choice test screen, the prompt is often your source
     * language and the target-language word only exists inside the answer
     * buttons - unreadable until you actually answer, since we can't know
     * which one is correct beforehand (and won't guess: reading a random
     * button could speak the wrong word entirely).
     *
     * Once you answer, Memrise highlights the correct button. Its utility
     * class names are hashed per-build (e.g. "sc-klj6s2-0 bbTWUX") so they
     * can't be pattern-matched reliably, but the highlight color itself is a
     * stable visual signal - so this checks each answer button's computed
     * background color for a green tint and reads that button's text once
     * one appears. This is a best-effort heuristic I haven't been able to
     * confirm against the actual "answer revealed" markup - if it doesn't
     * fire, send me that screen's HTML (right after you pick correctly) and
     * I'll swap this for something exact.
     */
    readRevealedAnswerDetail: function () {
      var card = document.querySelector('[data-testid="testLearnableCard"]');
      if (!card) { return null; }

      var buttons = card.querySelectorAll('button');
      for (var i = 0; i < buttons.length; i++) {
        var btn = buttons[i];
        var bg;
        try { bg = window.getComputedStyle(btn).backgroundColor || ""; }
        catch (e) { continue; }

        var m = /rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/.exec(bg);
        if (!m) { continue; }
        var r = +m[1], g = +m[2], b = +m[3];
        var looksGreen = g > r + 20 && g > b + 20;
        if (!looksGreen) { continue; }

        var text = (btn.textContent || "").trim();
        // The button also contains the "1"/"2"/"3"/"4" index label - strip a
        // single leading digit if present so it isn't spoken.
        text = text.replace(/^\d+/, "").trim();
        if (text) { return { text: text, element: btn }; }
      }
      return null;
    },

    readRevealedAnswer: function () {
      var d = this.readRevealedAnswerDetail();
      return d ? d.text : null;
    },

    /**
     * The "you got it wrong, here's the correct answer" screen. Answering
     * incorrectly used to leave the new/correct word unspoken entirely,
     * because readRevealedAnswer() only scans *clickable* multiple-choice
     * buttons, and readTypedAnswer() only fires when your own typed input
     * turned green (i.e. you were right). On a wrong answer neither of those
     * matches, even though the correct word is now shown on screen.
     *
     * This is a second, best-effort pass for that case:
     *  1) any element inside a *LearnableCard that's highlighted the same
     *     "correct" green used elsewhere on these screens - covers cards
     *     that show a static (non-button) reveal rather than a clickable one.
     *  2) a handful of testid patterns some Memrise builds use for this kind
     *     of correction/reveal text.
     * I don't have confirmed markup for this exact screen - if it still
     * doesn't autoplay after an incorrect answer, send me the HTML of that
     * screen and I'll wire up the precise selector.
     */
    readWrongAnswerCorrectionDetail: function () {
      var cards = document.querySelectorAll('[data-testid$="LearnableCard"]');
      for (var c = 0; c < cards.length; c++) {
        var candidates = cards[c].querySelectorAll('button, li, [role="option"], [role="button"]');
        for (var i = 0; i < candidates.length; i++) {
          var el = candidates[i];
          var bg;
          try { bg = window.getComputedStyle(el).backgroundColor || ""; }
          catch (e) { continue; }

          var m = /rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/.exec(bg);
          if (!m) { continue; }
          var r = +m[1], g = +m[2], b = +m[3];
          if (!(g > r + 20 && g > b + 20)) { continue; }

          var text = (el.textContent || "").trim().replace(/^\d+/, "").trim();
          if (text && text.length < 120) { return { text: text, element: el }; }
        }
      }

      var sel = '[data-testid$="answer-text"], [data-testid$="correct-text"],'
        + ' [data-testid*="correction"], [data-testid*="correctAnswer"]';
      try {
        var els = document.querySelectorAll(sel);
        for (var j = 0; j < els.length; j++) {
          var t = (els[j].textContent || "").trim();
          if (t && t.length < 120) { return { text: t, element: els[j] }; }
        }
      } catch (e) { /* ignore */ }

      return null;
    },

    readWrongAnswerCorrection: function () {
      var d = this.readWrongAnswerCorrectionDetail();
      return d ? d.text : null;
    },

    /**
     * The "type the correct translation" screen (confirmed: testid
     * typing-response-input) shows the target-language name as a plain
     * label near the input - "Dutch" in the sample - but the word itself
     * genuinely isn't in the DOM until you type it, correctly or not, so
     * there's nothing to read before you answer.
     *
     * Unlike the multiple-choice screen, this one hands us a real language
     * label to check against (not just a color guess), so first confirm the
     * input's own label actually matches the target language before trusting
     * it at all. Only once that's confirmed do we also check for a green
     * tint (the same best-effort "looks correct" signal as
     * readRevealedAnswer) before reading it aloud - otherwise it would speak
     * every half-finished guess while you're still typing. I don't have
     * confirmed markup for the "correct" state on this screen either - if
     * this never fires, send the HTML right after you type it correctly.
     */
    readTypedAnswerDetail: function () {
      var input = document.querySelector('[data-testid="typing-response-input"]');
      if (!input || !input.value) { return null; }

      // The input's own label names the language it expects. That's a real
      // label, not a guess - so it decides the side outright. Previously a
      // non-target label made this return null and the text fell through to
      // the generic prompt path, where it could be read with the wrong voice.
      var langName = this.findLanguageLabelNear(input);
      var side = langName ? this.sideOfLanguage(langName) : null;

      var wrapper = input.parentElement;
      var looksGreen = [input, wrapper].filter(Boolean).some(function (el) {
        var s;
        try { s = window.getComputedStyle(el); } catch (e) { return false; }
        var color = s.borderColor || s.backgroundColor || "";
        var m = /rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/.exec(color);
        if (!m) { return false; }
        var r = +m[1], g = +m[2], b = +m[3];
        return g > r + 20 && g > b + 20;
      });
      if (!looksGreen) { return null; }

      var value = input.value.trim();
      if (!value) { return null; }
      return { text: value, side: side, element: input };
    },

    readTypedAnswer: function () {
      var d = this.readTypedAnswerDetail();
      return d ? d.text : null;
    },

    /**
     * The visible prompt element, via the configurable selector. Returned as
     * an element (not just text) so its on-screen language label can be
     * inspected - the prompt is NOT always the target language.
     */
    findPromptElement: function () {
      try {
        var els = document.querySelectorAll(this.promptSelector);
        for (var i = 0; i < els.length; i++) {
          var t = (els[i].textContent || "").trim();
          if (t && t.length < 120) { return els[i]; }
        }
      } catch (e) {
        this.lastError = "bad prompt selector";
      }
      return null;
    },

    /**
     * Works out WHICH LANGUAGE each piece of text on screen is, instead of
     * assuming the prompt is the target language. A Memrise card can be
     * pointed either way round: sometimes the question is the target
     * language and the answer your own, sometimes the reverse.
     *
     * Returns { target, source, pending, evidence }:
     *   target  - the target-language text on screen, or null if there
     *             isn't one yet (e.g. it's still hidden behind an answer)
     *   source  - the source-language text, or null
     *   pending - true when this screen can't reveal the target word until
     *             you answer, so nothing should be spoken or offered yet
     *   evidence- how each decision was reached, for diagnose()
     *
     * Evidence is used strongest-first:
     *   1. a card that labels each side with its language (definitive)
     *   2. a language label displayed near the text (definitive)
     *   3. the writing system - Cyrillic/kana/Greek/etc (definitive)
     *   4. stopwords, checked against BOTH languages (strong)
     *   5. the two sides of a card are always opposite languages, so a
     *      known side settles the other one
     *   6. only if all of the above fail: assume the prompt is the target
     */
    resolveScreenTexts: function () {
      var result = { target: null, source: null, pending: false, evidence: [] };

      // (1) A card that names each side's language outright.
      var pairs = this.readCardWithLanguageLabels();
      if (pairs && (pairs.target || pairs.source)) {
        result.target = pairs.target || null;
        result.source = pairs.source || null;
        result.evidence.push("card language labels");
        return result;
      }

      // (2) The answer side - only ever readable once actually answered.
      var answerText = null, answerSide = null, answerEl = null;
      var typed = this.readTypedAnswerDetail();
      if (typed) {
        answerText = typed.text;
        answerSide = typed.side;
        answerEl   = typed.element;
        if (answerSide) { result.evidence.push("typing input label -> answer is " + answerSide); }
      } else {
        var revealed = this.readRevealedAnswerDetail() || this.readWrongAnswerCorrectionDetail();
        if (revealed) {
          answerText = revealed.text;
          answerEl   = revealed.element;
        }
      }
      if (answerEl && !answerSide) {
        var answerLabel = this.findLanguageLabelNear(answerEl);
        if (answerLabel) {
          answerSide = this.sideOfLanguage(answerLabel);
          if (answerSide) { result.evidence.push("label near answer -> " + answerSide); }
        }
      }

      // (3) The prompt side.
      var promptEl   = this.findPromptElement();
      var promptText = promptEl ? (promptEl.textContent || "").trim() : "";
      var promptSide = null;
      if (promptEl) {
        var promptLabel = this.findLanguageLabelNear(promptEl);
        if (promptLabel) {
          promptSide = this.sideOfLanguage(promptLabel);
          if (promptSide) { result.evidence.push("label near prompt -> " + promptSide); }
        }
      }

      // Screens where the target word simply isn't in the DOM until answered.
      var isAnswerScreen = !!document.querySelector(
        '[data-testid="testLearnableCard"], [data-testid="typing-response-input"]');
      result.pending = isAnswerScreen && !answerText;

      // (4) Script and stopwords.
      if (!promptSide && promptText) {
        promptSide = this.classifyText(promptText);
        if (promptSide) { result.evidence.push("prompt text looks like " + promptSide); }
      }
      if (!answerSide && answerText) {
        answerSide = this.classifyText(answerText);
        if (answerSide) { result.evidence.push("answer text looks like " + answerSide); }
      }

      // (5) The two sides of one card are always opposite languages.
      if (promptText && answerText && promptText !== answerText) {
        if (promptSide && !answerSide) {
          answerSide = this.otherSide(promptSide);
          result.evidence.push("answer is opposite of prompt -> " + answerSide);
        } else if (answerSide && !promptSide) {
          promptSide = this.otherSide(answerSide);
          result.evidence.push("prompt is opposite of answer -> " + promptSide);
        }
      }

      // (6) Genuinely no evidence left (both languages Latin-script, no
      // labels, no stopword hits). Fall back to the historic assumption.
      if (promptText && !promptSide) {
        promptSide = "target";
        result.evidence.push("no evidence - assumed prompt is target");
      }
      if (answerText && !answerSide) {
        answerSide = promptText ? this.otherSide(promptSide) : "target";
        result.evidence.push("no evidence - assumed answer is " + answerSide);
      }

      var assign = function (text, side) {
        if (!text || !side) { return; }
        if (side === "target" && !result.target) { result.target = text; }
        if (side === "source" && !result.source) { result.source = text; }
      };
      // The answer is the more specific reading, so it claims its side first.
      assign(answerText, answerSide);
      assign(promptText, promptSide);

      return result;
    },

    /**
     * Reads the current screen and plays what should be played.
     * force=true re-reads and re-plays even if nothing changed (used when
     * the user flips a setting).
     *
     * The rules, in full:
     *  - target-language text is always spoken with the TARGET voice
     *  - source-language text is only spoken at all when opted in, and then
     *    always with the SOURCE voice
     *  - text whose language can't be established is never spoken with a
     *    voice that wasn't chosen for it
     */
    readPromptFromDom: function (force) {
      var courseId = this.currentCourseId();
      if (courseId !== this.domCourseId) {
        this.applyLanguage();
        this.editDomAudioOptions();
      }

      if (this.wordColumn === NONE) {
        this.currentWord = "";
        this.currentWordLang = null;
        this.skippedAsSource = false;
        this.answerPending = false;
        this.refreshStatus();
        this.updatePlayAgainButtons();
        return "";
      }

      var screen = this.resolveScreenTexts();
      this.lastEvidence  = screen.evidence.join(" | ");
      this.answerPending = screen.pending;

      var targetText = screen.target || "";
      var sourceText = screen.source || "";
      var mayPlaySource = this.mayPlaySourceLanguage();

      // What the "play again" button replays. The target-language text when
      // there is one; the source text only when the user opted into hearing
      // their own language.
      var word = targetText || (mayPlaySource ? sourceText : "");
      var side = targetText ? "target" : (word ? "source" : null);

      this.currentWord     = word;
      this.currentWordLang = side ? this.langForSide(side) : null;
      // "skipped" = there was something on screen, but it was your own
      // language and you haven't asked to hear that.
      this.skippedAsSource = !targetText && !!sourceText && !mayPlaySource;

      this.refreshStatus();
      this.updatePlayAgainButtons();

      if (!word) { return ""; }

      var key = "t:" + targetText + "|s:" + (mayPlaySource ? sourceText : "");
      if (!force && key === this.lastPlayed) { return this.currentWord; }

      if (this.autoplay && this.shouldAutoplayGivenNativeAudio()) {
        this.lastPlayed = key;

        var queue = [];
        if (targetText) { queue.push({ word: targetText, lang: this.language }); }
        if (mayPlaySource && sourceText && sourceText !== targetText) {
          queue.push({ word: sourceText, lang: this.sourceLanguage });
        }
        if (queue.length) { this.speakSequence(queue); }
      }
      return this.currentWord;
    },

    //+------------------------------------------------------
    //| GENERATE AUDIO
    //+------------------------------------------------------

    playGeneratedAudio: function (word, tried, lang, onEnded) {
      tried = tried || {};
      lang  = lang || this.language;
      if (!word) {
        this.setStatus("No word to play - check the column/selector setting");
        if (onEnded) { onEnded(); }
        return;
      }

      // HARD GATE - last line of defence, checked at the one place audio
      // actually gets generated, so it can't be bypassed by a bug anywhere
      // upstream (a mislabelled column, a screen the heuristics don't fully
      // recognise, a missed branch). Blocks both "this call asked for the
      // source voice" and "this text is source-language text regardless of
      // what voice was requested".
      if (!this.mayPlaySourceLanguage()) {
        var langIsSource = !!lang && !!this.sourceLanguage && lang === this.sourceLanguage
          && this.language !== this.sourceLanguage;
        if (langIsSource || this.classifyText(word) === "source") {
          log("blocked: source-language content, and hearing it isn't enabled (\"" + word + "\")");
          if (onEnded) { onEnded(); }
          return;
        }
      }

      // Never speak text in a voice that wasn't chosen for its language: if
      // this call asked for the target voice but the text is clearly source
      // text (or vice versa), correct the voice rather than mispronouncing it.
      var actualSide = this.classifyText(word);
      if (actualSide) {
        var correctLang = this.langForSide(actualSide);
        if (correctLang && lang !== correctLang) {
          log("voice corrected: \"" + word + "\" is " + actualSide
            + " text, using " + correctLang + " not " + lang);
          lang = correctLang;
        }
      }

      var order = this.engineOrderFor(lang);
      for (var i = 0; i < order.length; i++) {
        var engine = order[i];
        if (tried[engine]) { continue; }

        if (engine === "fish")   { this.fetchFishAudio(word, true, tried, lang, onEnded); return; }
        if (engine === "google") { this.fetchGoogleTts(word, true, tried, lang, onEnded); return; }
        if (engine === "speech") {
          tried.speech = true;
          this.playSpeechSynthesisAudio(word, lang, onEnded);
          return;
        }
        if (engine === "voicerss") {
          tried.voicerss = true;
          this.playVoiceRssAudio(word, tried, lang, onEnded);
          return;
        }
      }

      if (!this.lastError) {
        this.lastError = lang ? "no voice for " + lang
          : "no voice available (language unknown - try picking one in the dropdown)";
      }
      log("no playable sources left for word: " + word);
      this.refreshStatus();
      if (onEnded) { onEnded(); }
    },

    /**
     * Which engines can speak `lang`, best first.
     *
     * The important part: Fish Audio has NO language parameter. It renders
     * whatever voice it's given, so asking it for a Dutch word without a
     * Dutch voice id gets you the word read in some other accent - which is
     * why audio was wrong so often, given Fish was unconditionally tried
     * first for every language. So Fish leads only when it's actually been
     * pointed at a voice for this language (or you've ticked "prefer Fish"),
     * and otherwise sits behind the engines that do take a language code.
     */
    engineOrderFor: function (lang) {
      var languageAware = [];
      if (GOOGLETTS_LANG[lang])                    { languageAware.push("google"); }
      if (this.getSpeechSynthesisVoiceFor(lang))   { languageAware.push("speech"); }
      if (this.voiceRssKey && VOICERSS_LANG[lang]) { languageAware.push("voicerss"); }

      var order = [];
      var hasFish = !!this.fishKey;
      var fishLeads = hasFish
        && (!!this.fishVoiceIdFor(lang) || this.preferFish || !languageAware.length);

      if (fishLeads) { order.push("fish"); }
      order = order.concat(languageAware);
      if (hasFish && order.indexOf("fish") === -1) { order.push("fish"); }

      return order;
    },

    /**
     * Plays a list of {word, lang} entries strictly one after another (never
     * overlapping), used when both the source and target language need to be
     * read for the same card. Each step waits for actual playback to finish
     * (via the 'ended'/onend event, not a guessed delay) before starting the
     * next, so there's never audio clash between them.
     */
    speakSequence: function (items, index) {
      index = index || 0;
      if (index >= items.length) { return; }
      var item = items[index];
      if (!item || !item.word) { this.speakSequence(items, index + 1); return; }
      this.playGeneratedAudio(item.word, {}, item.lang, function () {
        AudioProvider.speakSequence(items, index + 1);
      });
    },

    /**
     * Stops whatever audio (HTMLAudioElement or speechSynthesis) is currently
     * playing before starting something new - without this, rapid card
     * changes or a source+target sequence could overlap into a garbled mess.
     */
    stopActiveAudio: function () {
      if (this.activeAudioElement) {
        try { this.activeAudioElement.pause(); } catch (e) { /* ignore */ }
      }
      try { window.speechSynthesis.cancel(); } catch (e) { /* ignore */ }
    },

    getSpeechSynthesisVoiceFor: function (lang) {
      var langCode = SPEECHSYNTHESIS_LANG[lang];
      if (!langCode || !this.speechSynthesisSupported) { return null; }

      var voices = [];
      try { voices = window.speechSynthesis.getVoices() || []; } catch (e) { voices = []; }

      var normalise = function (l) { return String(l || "").replace(/_/g, "-").toLowerCase(); };
      var wanted = normalise(langCode), prefix = wanted.split("-")[0];

      var voice = U.find(voices, function (v) { return normalise(v.lang) === wanted; })
        || U.find(voices, function (v) { return normalise(v.lang).split("-")[0] === prefix; });

      return voice ? { voice: voice, langCode: langCode } : null;
    },

    playSpeechSynthesisAudio: function (word, lang, onEnded) {
      this.stopActiveAudio();
      var picked = this.getSpeechSynthesisVoiceFor(lang);
      if (!picked) { if (onEnded) { onEnded(); } return; }

      log("speaking with browser voice (" + lang + "): " + word);

      var utter = new window.SpeechSynthesisUtterance();
      utter.lang  = picked.langCode;
      utter.voice = picked.voice;
      utter.text  = word;

      var fired = false;
      var finish = function () { if (!fired) { fired = true; if (onEnded) { onEnded(); } } };
      utter.onend   = finish;
      utter.onerror = finish;

      window.speechSynthesis.speak(utter);
    },

    /**
     * Ask the sandbox half to make a cross-origin request for us and hand back
     * a data: url. Resolves/rejects; times out so a missing bridge degrades to
     * the next audio source instead of hanging forever.
     */
    bridgeRequest: function (options) {
      return new Promise(function (resolve, reject) {
        var id = "r" + (++AudioProvider.bridgeSeq) + "_" + Date.now();

        var timer = setTimeout(function () {
          document.removeEventListener(BRIDGE_RESPONSE + id, onResponse);
          reject(new Error("no userscript bridge (needs GM_xmlhttpRequest)"));
        }, AudioProvider.bridgeTimeout || BRIDGE_TIMEOUT_MS);

        var onResponse = function (e) {
          clearTimeout(timer);
          document.removeEventListener(BRIDGE_RESPONSE + id, onResponse);

          var data;
          try { data = JSON.parse(e.detail); }
          catch (err) { reject(new Error("bad bridge response")); return; }

          if (data.ok) { resolve(data); } else { reject(new Error(data.error || "request failed")); }
        };

        document.addEventListener(BRIDGE_RESPONSE + id, onResponse);
        document.dispatchEvent(new CustomEvent(BRIDGE_REQUEST, {
          detail: JSON.stringify({
            id: id, url: options.url, method: options.method || "GET",
            headers: options.headers || {}, body: options.body || null
          })
        }));
      });
    },

    /**
     * Fish voice ids are stored per LANGUAGE. Storing them per course was
     * itself a source of wrong audio: one course contains two languages, so
     * a voice chosen for the target language was also used to read the
     * source language, and vice versa.
     */
    fishVoiceIdFor: function (lang) {
      if (!lang) { return ""; }
      return this.fishVoicesByLang[lang] || "";
    },

    fishVoiceId: function () {
      return this.fishVoiceIdFor(this.language);
    },

    refreshFishVoiceField: function () {
      var field = document.getElementById('audio-provider-fishvoice');
      var label = document.getElementById('audio-provider-fishvoice-label');
      if (label) {
        label.textContent = this.language
          ? "Fish voice id for " + this.language + " (optional):"
          : "Fish voice id (optional):";
      }
      if (field) {
        field.value = this.fishVoiceIdFor(this.language);
        field.disabled = !this.language;
      }
    },

    /**
     * Fish Audio. Voice-driven rather than language-driven, so this works even
     * when the course category gives us no usable language.
     */
    fetchFishAudio: function (word, play, tried, lang, onEnded) {
      lang = lang || this.language;
      var voiceId = this.fishVoiceIdFor(lang);
      // Cache key carries the language AND the voice: the same word can
      // legitimately appear on both sides of a course, and cached audio from
      // one language must never be replayed for the other.
      var source = "fish audio:" + (lang || "?") + ":" + (voiceId || "default"),
          cached = this.getCachedElement(source, word);
      if (cached) {
        if (play) { this.playElement(cached, onEnded); } else if (onEnded) { onEnded(); }
        return;
      }
      if (this.inflight[source + word]) { if (onEnded) { onEnded(); } return; }
      this.inflight[source + word] = true;

      var body = { text: word, format: "mp3", mp3_bitrate: 128, latency: "normal" };
      if (voiceId) { body.reference_id = voiceId; }

      log("fetching fish audio (" + this.fishModel + ", " + (lang || "no language")
        + ", voice " + (voiceId || "default") + ") for: " + word);

      this.bridgeRequest({
        url: FISH_TTS_URL,
        method: "POST",
        headers: {
          "Authorization": "Bearer " + this.fishKey,
          "Content-Type": "application/json",
          "model": this.fishModel
        },
        body: JSON.stringify(body)
      }).then(function (res) {
        delete AudioProvider.inflight[source + word];

        var audioElement = document.createElement('audio');
        audioElement.__audioProviderOwned = true;
        audioElement.src = res.dataUrl;
        AudioProvider.setCachedElement(source, word, audioElement);
        AudioProvider.lastError = null;
        delete AudioProvider.sourceErrors.fish;
        AudioProvider.refreshStatus();
        if (play) { AudioProvider.playElement(audioElement, onEnded); } else if (onEnded) { onEnded(); }

      }).catch(function (e) {
        delete AudioProvider.inflight[source + word];
        // Record the failure for diagnostics, but DON'T disable Fish for the
        // rest of the session - a single rate-limited/timed-out request used
        // to lock out every future word, which is why only the first word
        // ever played. Just fall through to the next source for this word;
        // the next word gets a fresh attempt at Fish again.
        AudioProvider.sourceErrors.fish = e.message;
        AudioProvider.lastError = "Fish: " + e.message;
        log("fish audio failed (" + e.message + ") - falling back for this word");
        AudioProvider.refreshStatus();
        if (play) {
          tried = tried || {};
          tried.fish = true;
          AudioProvider.playGeneratedAudio(word, tried, lang, onEnded);
        } else if (onEnded) { onEnded(); }
      });
    },

    /**
     * Google sends no CORS headers, so go through the proxy, take the mp3 as a
     * blob and play that. No referrer tricks, no cross-origin media element.
     */
    fetchGoogleTts: function (word, play, tried, lang, onEnded) {
      lang = lang || this.language;
      var languageCode = GOOGLETTS_LANG[lang];
      if (!languageCode) { if (onEnded) { onEnded(); } return; }

      var source = "google tts:" + lang,
          cached = this.getCachedElement(source, word);
      if (cached) {
        if (play) { this.playElement(cached, onEnded); } else if (onEnded) { onEnded(); }
        return;
      }
      if (this.pending === source + word) { if (onEnded) { onEnded(); } return; }
      this.pending = source + word;

      var url = this.getGoogleTtsUrl(languageCode, word);
      log("fetching google tts (" + lang + ") for: " + word);

      fetch(url)
        .then(function (res) {
          if (!res.ok) { throw new Error("HTTP " + res.status); }
          return res.blob();
        })
        .then(function (blob) {
          AudioProvider.pending = null;
          if (!blob || blob.size < 512) { throw new Error("empty audio response"); }

          var audioElement = document.createElement('audio');
          audioElement.__audioProviderOwned = true;
          audioElement.src = URL.createObjectURL(blob);
          AudioProvider.setCachedElement(source, word, audioElement);
          AudioProvider.lastError = null;
          delete AudioProvider.sourceErrors.google;
          AudioProvider.refreshStatus();
          if (play) { AudioProvider.playElement(audioElement, onEnded); } else if (onEnded) { onEnded(); }
        })
        .catch(function (e) {
          AudioProvider.pending = null;
          AudioProvider.sourceErrors.google = e.message;
          AudioProvider.lastError = "Google TTS: " + e.message;
          log("google tts failed (" + e.message + ") - falling back for this word");
          AudioProvider.refreshStatus();
          if (play) {
            tried = tried || {};
            tried.google = true;
            AudioProvider.playGeneratedAudio(word, tried, lang, onEnded);
          } else if (onEnded) { onEnded(); }
        });
    },

    getGoogleTtsUrl: function (languageCode, word) {
      var google = "https://translate.google.com/translate_tts?ie=UTF-8"
        + "&tl=" + (languageCode || "en")
        + "&client=tw-ob&q=" + encodeURIComponent(word)
        + "&tk=" + Math.floor(Math.random() * 1000000)
        + "&ttsspeed=1";
      return TTS_PROXY + encodeURIComponent(google);
    },

    playVoiceRssAudio: function (word, tried, lang, onEnded) {
      lang = lang || this.language;
      var languageCode = VOICERSS_LANG[lang];
      if (!languageCode) { if (onEnded) { onEnded(); } return; }

      var source = "voice rss:" + lang,
          cached = this.getCachedElement(source, word);
      if (cached) { this.playElement(cached, onEnded); return; }

      log("fetching voice rss (" + lang + ") for: " + word);

      var audioElement = document.createElement('audio');
      audioElement.__audioProviderOwned = true;
      audioElement.src = this.getVoiceRssUrl(languageCode, word);
      audioElement.addEventListener('error', function () {
        AudioProvider.sourceErrors.voicerss = "request failed";
        AudioProvider.lastError = "Voice RSS request failed";
        AudioProvider.removeCachedElement(source, word);
        AudioProvider.refreshStatus();
        if (onEnded) { onEnded(); }
      });
      this.setCachedElement(source, word, audioElement);
      this.playElement(audioElement, onEnded);
    },

    getVoiceRssUrl: function (languageCode, word) {
      return `https://api.voicerss.org/?key=${this.voiceRssKey}&src=${encodeURIComponent(word)}&hl=${languageCode}&f=48khz_16bit_stereo`;
    },

    /**
     * Browsers reject play() without a user gesture; surface that instead of
     * failing silently. Also stops any currently-playing audio first (see
     * stopActiveAudio) so a new word can never overlap with the previous one,
     * and calls onEnded (if given) once playback genuinely finishes, so a
     * source+target sequence never overlaps either.
     */
    playElement: function (audioElement, onEnded) {
      this.stopActiveAudio();
      this.activeAudioElement = audioElement;

      try {
        audioElement.currentTime = 0;
      } catch (e) { /* not seekable yet */ }

      if (onEnded) {
        var fired = false;
        var finish = function () { if (!fired) { fired = true; onEnded(); } };
        audioElement.addEventListener('ended', finish, { once: true });
        audioElement.addEventListener('error', finish, { once: true });
        // Safety net in case neither event ever fires for some reason.
        setTimeout(finish, 8000);
      }

      var result = audioElement.play();
      if (result && typeof result.catch === "function") {
        result.catch(function (e) {
          AudioProvider.lastError = "playback blocked (" + e.name + ") - click the page once";
          log("playback rejected: " + e.message);
          AudioProvider.refreshStatus();
          if (onEnded) { onEnded(); }
        });
      }
    },

    //+------------------------------------------------------
    //| CACHE
    //+------------------------------------------------------

    getCachedElement: function (source, word) {
      var cached = U.find(this.cachedAudioElements, function (obj) {
        return obj.source === source && obj.word === word;
      });
      return cached && cached.element;
    },

    removeCachedElement: function (source, word) {
      U.remove(this.cachedAudioElements, function (e) {
        return e.source === source && e.word === word;
      });
    },

    removeCachedSource: function (source) {
      U.remove(this.cachedAudioElements, function (e) { return e.source === source; });
    },

    /**
     * Cache keys now carry the language and voice (e.g. "fish audio:Dutch:abc123"),
     * so clearing "all Fish audio" has to match on the prefix.
     */
    removeCachedSourcePrefix: function (prefix) {
      U.remove(this.cachedAudioElements, function (e) {
        return String(e.source).indexOf(prefix) === 0;
      });
    },

    setCachedElement: function (source, word, element) {
      this.cachedAudioElements.push({ source: source, word: word, element: element });
      if (this.cachedAudioElements.length > 60) {
        this.cachedAudioElements.shift();
      }
    }
  };

  function log(message) {
    console.log("Audio Provider: " + message);
  }

  window.MemriseAudioProvider = AudioProvider;

  function bootstrap() {
    var tries = 0;
    var timer = setInterval(function () {
      tries++;
      // Start as soon as the page is interactive; the session watcher and DOM
      // fallback take it from there whether or not MEMRISE ever shows up.
      if (document.body) {
        clearInterval(timer);
        AudioProvider.init();
      } else if (tries > 120) {
        clearInterval(timer);
      }
    }, 250);
  }

  if (document.readyState === "complete" || document.readyState === "interactive") {
    bootstrap();
  } else {
    window.addEventListener('load', bootstrap, false);
  }
}

// ---- BRIDGE (sandbox half) ----------------------------------------------
// Runs where GM_xmlhttpRequest exists. Performs cross-origin requests the page
// cannot make itself and returns the audio as a data: url over a CustomEvent.

function installBridge() {
  const BRIDGE_REQUEST  = "memrise-audio-provider-request",
        BRIDGE_RESPONSE = "memrise-audio-provider-response";

  var xhr = typeof GM_xmlhttpRequest === "function" ? GM_xmlhttpRequest
    : (typeof GM !== "undefined" && GM && GM.xmlHttpRequest ? GM.xmlHttpRequest.bind(GM) : null);

  if (!xhr) {
    console.log("Audio Provider: no GM_xmlhttpRequest - Fish Audio will be unavailable");
    return false;
  }

  var reply = function (id, payload) {
    document.dispatchEvent(new CustomEvent(BRIDGE_RESPONSE + id, {
      detail: JSON.stringify(payload)
    }));
  };

  document.addEventListener(BRIDGE_REQUEST, function (e) {
    var req;
    try { req = JSON.parse(e.detail); } catch (err) { return; }

    xhr({
      method: req.method || "GET",
      url: req.url,
      headers: req.headers || {},
      data: req.body || undefined,
      responseType: "blob",
      timeout: 20000,

      onload: function (res) {
        // Errors come back as JSON, not audio - surface the real message
        // (e.g. 402 Insufficient API credit) instead of playing silence.
        if (res.status < 200 || res.status > 299) {
          var reader = new FileReader();
          reader.onloadend = function () {
            var msg = String(reader.result || "");
            try { msg = JSON.parse(msg).message || msg; } catch (err) { /* not json */ }
            reply(req.id, { ok: false, error: "HTTP " + res.status + " " + msg.slice(0, 120) });
          };
          reader.onerror = function () {
            reply(req.id, { ok: false, error: "HTTP " + res.status });
          };
          reader.readAsText(res.response);
          return;
        }

        var blob = res.response;
        if (!blob || blob.size < 512) {
          reply(req.id, { ok: false, error: "empty audio response" });
          return;
        }
        var reader2 = new FileReader();
        reader2.onloadend = function () {
          reply(req.id, { ok: true, status: res.status, dataUrl: String(reader2.result) });
        };
        reader2.onerror = function () {
          reply(req.id, { ok: false, error: "could not read audio" });
        };
        reader2.readAsDataURL(blob);
      },

      onerror:   function ()  { reply(req.id, { ok: false, error: "network error" }); },
      ontimeout: function ()  { reply(req.id, { ok: false, error: "timed out" }); },
      onabort:   function ()  { reply(req.id, { ok: false, error: "aborted" }); }
    });
  });

  // Let the page half report bridge presence in diagnose()
  try {
    var flag = document.createElement('script');

  return true;
}
```
